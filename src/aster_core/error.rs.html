159/// An error with an error number and a source location.
160///
161/// The location identifies where this error was constructed,
162/// including where an external error was converted into this type.
163/// Propagating an existing error preserves its original location.
164#[derive(Clone, Copy, Debug)]
165pub struct Error {
166    errno: Errno,
167    location: &'static Location<'static>,
168}
169
170impl Error {
171    /// Creates an error with the given error number and the caller's source location.
172    ///
173    /// Use [`errno!`](crate::errno) when the error number is a known variant.
174    /// Call this method directly when the error number is computed at runtime.
175    #[track_caller]
176    pub const fn new(errno: Errno) -> Self {
177        Error {
178            errno,
179            location: Location::caller(),
180        }
181    }
182
183    /// Returns the location where this error was constructed.
184    ///
185    /// Propagating the error preserves this location.
186    pub const fn location(&self) -> &'static Location<'static> {
187        self.location
188    }
189
190    /// Returns the error number.
191    pub const fn error(&self) -> Errno {
192        self.errno
193    }
194}
195
196/// Constructs a kernel error at the call site from an `Errno` variant name.
197///
198/// The optional message documents the error in the source; it is neither
199/// evaluated nor stored in the error.
200#[macro_export]
201macro_rules! errno {
202    ($errno:ident $(, $message:expr)? $(,)?) => {
203        $crate::error::Error::new($crate::error::Errno::$errno)
204    };
205}

220impl From<ostd::Error> for Error {
221    #[track_caller]
222    fn from(ostd_error: ostd::Error) -> Self {
223        match ostd_error {
224            ostd::Error::AccessDenied => errno!(EFAULT),
225            ostd::Error::NoMemory => errno!(ENOMEM),
226            ostd::Error::InvalidArgs => errno!(EINVAL),
227            ostd::Error::IoError => errno!(EIO),
228            ostd::Error::NotEnoughResources => errno!(EBUSY),
229            ostd::Error::PageFault => errno!(EFAULT),
230            ostd::Error::Overflow => errno!(EOVERFLOW),
231        }
232    }

393impl From<aster_device::common::Error> for Error {
394    #[track_caller]
395    fn from(err: aster_device::common::Error) -> Self {
396        use aster_device::common::Error::*;
397        match err {
398            AlreadyAdded | AlreadyBound => errno!(EBUSY),
399            NotAdded | NotBound | ParentNotAdded | DriverUnregistered | NoDevNum => {
400                errno!(ENODEV)
401            }
402            HasChildren => errno!(ENOTEMPTY),
403            NameConflict => errno!(EEXIST),
404            NotFound | NoDriver => errno!(ENOENT),
405            InvalidName | InvalidValue => errno!(EINVAL),
406            ProbeFailed | Attribute | Format | Hook => errno!(EIO),
407            ResourceUnavailable => errno!(ENOSPC),
408            MemoryBudgetExhausted => errno!(ENOMEM),
409            SysTree(inner) => inner.into(),
410        }
411    }
412}

445#[cfg(ktest)]
446mod tests {
447    use ostd::prelude::*;
448
449    use super::*;
450
451    #[ktest]
452    fn construction_location() {
453        let line = line!() + 1;
454        let error = Error::new(Errno::EINVAL);
455        assert_eq!(error.location().file(), file!());
456        assert_eq!(error.location().line(), line);
457
458        let line = line!() + 1;
459        let error = errno!(EINVAL, "invalid argument");
460        assert_eq!(error.error(), Errno::EINVAL);
461        assert_eq!(error.location().line(), line);
462
463        let line = line!() + 1;
464        const ERROR: Error = errno!(EINVAL,);
465        assert_eq!(ERROR.error(), Errno::EINVAL);
466        assert_eq!(ERROR.location().line(), line);
467
468        let line = line!() + 2;
469        fn fail() -> Result<(), Error> {
470            Err(errno!(EINVAL))
471        }
472        assert_eq!(fail().unwrap_err().location().line(), line);
473
474        let line = line!() + 2;
475        fn fail_with_message() -> Result<(), Error> {
476            Err(errno!(EINVAL, "invalid argument"))
477        }
478        assert_eq!(fail_with_message().unwrap_err().location().line(), line);
479    }
480
481    #[ktest]
482    fn messages_are_not_evaluated() {
483        const ERROR: Error = errno!(EINVAL, panic!("message must not be evaluated"),);
484        assert_eq!(ERROR.error(), Errno::EINVAL);
485
486        fn fail() -> Result<(), Error> {
487            Err(errno!(EIO, panic!("message must not be evaluated"),))
488        }
489        assert_eq!(fail().unwrap_err().error(), Errno::EIO);
490    }
491
492    #[ktest]
493    fn conversion_location() {
494        let line = line!() + 1;
495        let error = Error::from(Errno::EINVAL);
496        assert_eq!(error.location().line(), line);
497
498        let line = line!() + 1;
499        let error: Error = ostd::Error::InvalidArgs.into();
500        assert_eq!(error.location().line(), line);
501
502        let line = line!() + 1;
503        let error = Error::from((ostd::Error::PageFault, 0));
504        assert_eq!(error.location().line(), line);
505
506        let line = line!() + 2;
507        fn convert() -> Result<(), Error> {
508            Err(ostd::Error::InvalidArgs)?;
509            Ok(())
510        }
511        let error = convert().unwrap_err();
512        assert_eq!(error.location().file(), file!());
513        assert_eq!(error.location().line(), line);
514    }
515
516    #[ktest]
517    fn propagation_preserves_location() {
518        fn propagate(error: Error) -> Result<(), Error> {
519            Err(error)?;
520            Ok(())
521        }
522
523        let original = errno!(EINVAL);
524        let propagated = propagate(original).unwrap_err();
525        assert_eq!(propagated.location(), original.location());
526
527        let unpacked = Error::from((original, 0));
528        assert_eq!(unpacked.location(), original.location());
529    }
