17pub(super) fn sys_set_priority(
18    which: i32,
19    who: u32,
20    prio: i32,
21    ctx: &Context,
22) -> Result<SyscallReturn> {
23    let prio_target = PriorityTarget::new(which, who, ctx)?;
24    let new_nice: Nice = {
25        let nice_raw = prio.clamp(
26            Nice::MIN.value().get() as i32,
27            Nice::MAX.value().get() as i32,
28        ) as i8;
29        nice_raw.try_into().unwrap()
30    };
31    debug!(
32        "set_priority: prio_target = {:?}, new_nice = {:?}",
33        prio_target, new_nice
34    );
35
36    let processes = get_processes(prio_target)?;
37    let mut error = None;
38    for process in processes.iter() {
39        if let Err(err) = set_priority_one(new_nice, process, ctx) {
40            error = Some(err);
41        }
42    }
43
44    if let Some(err) = error {
45        return Err(err);
46    }
47
48    Ok(SyscallReturn::Return(0))
49}
50
51fn set_priority_one(new_nice: Nice, process: &Process, ctx: &Context) -> Result<()> {
52    check_setpriority_permission(process, ctx)?;
53
54    let rlimit = process.resource_limits();
55    let limit = rlimit.get_rlimit(RLIMIT_NICE).get_cur();
56    if new_nice < process.nice().load(Ordering::Relaxed)
57        && (20 - new_nice.value().get()) as u64 > limit
58        && check_nice_capability(ctx).is_err()
59    {
60        return_errno_with_message!(Errno::EACCES, "the nice value reaches the resource limit");
61    }
62
63    // FIXME: `setpriority` updates only the per-process nice value. Fair
64    // scheduler state is kept in each thread's `SchedPolicy::Fair`, so it
65    // should be updated from the same source of truth.
66    process.nice().store(new_nice, Ordering::Relaxed);
67
68    Ok(())
69}

71/// Checks whether the caller may change the target process's nice value.
72fn check_setpriority_permission(target: &Process, ctx: &Context) -> Result<()> {
73    let caller_uid = ctx.posix_thread.credentials().euid();
74    let target_credentials = target
75        .main_thread()
76        .as_posix_thread()
77        .unwrap()
78        .credentials();
79
80    if caller_uid != target_credentials.ruid() && caller_uid != target_credentials.euid() {
81        lsm_hooks::on_capable(lsm_hooks::CapableContext::new(
82            target.user_ns().lock().as_ref(),
83            ctx.posix_thread,
84            CapSet::SYS_NICE,
85        ))?;
86    }
87    Ok(())
88}

90/// Checks whether the caller has the `SYS_NICE` capability in the initial user namespace.
91fn check_nice_capability(ctx: &Context) -> Result<()> {
92    lsm_hooks::on_capable(lsm_hooks::CapableContext::new(
93        UserNamespace::get_init_singleton().as_ref(),
94        ctx.posix_thread,
95        CapSet::SYS_NICE,
96    ))
97}
