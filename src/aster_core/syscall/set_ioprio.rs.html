12pub(super) fn sys_ioprio_set(
13    which: u32,
14    who: u32,
15    ioprio: u32,
16    ctx: &Context,
17) -> Result<SyscallReturn> {
18    debug!("which = {}, who = {}, ioprio = {}", which, who, ioprio);
19
20    check_ioprio_class(ioprio, ctx)?;
21
22    let ioprio_who = IoPrioWho::from_which_and_who(which, who, ctx)?;
23
24    match ioprio_who {
25        IoPrioWho::Thread(thread) => {
26            let target = thread.as_posix_thread().unwrap();
27            let target_uid = target.credentials().ruid();
28            let caller_credentials = ctx.posix_thread.credentials();
29
30            if target_uid != caller_credentials.euid() && target_uid != caller_credentials.ruid() {
31                lsm_hooks::on_capable(lsm_hooks::CapableContext::new(
32                    UserNamespace::get_init_singleton().as_ref(),
33                    ctx.posix_thread,
34                    CapSet::SYS_NICE,
35                ))?;
36            }
37
38            target.io_priority().store(ioprio, Ordering::Relaxed);
39
40            Ok(SyscallReturn::Return(0))
41        }
42        IoPrioWho::ProcessGroup(_) => {
43            // TODO: Implement process group support
44            Err(errno!(EINVAL))
45        }
46        IoPrioWho::User(_) => {
47            // TODO: Implement user support
48            Err(errno!(EINVAL))
49        }
50    }
51}

53/// Validates the I/O priority class and checks permission for real-time I/O.
54fn check_ioprio_class(ioprio: u32, ctx: &Context) -> Result<()> {
55    let class = IoPrioClass::try_from((ioprio >> IOPRIO_CLASS_SHIFT) & IOPRIO_CLASS_MASK)?;
56
57    match class {
58        IoPrioClass::RealTime => {
59            let user_ns = UserNamespace::get_init_singleton();
60
61            if lsm_hooks::on_capable(lsm_hooks::CapableContext::new(
62                user_ns.as_ref(),
63                ctx.posix_thread,
64                CapSet::SYS_ADMIN,
65            ))
66            .is_err()
67            {
68                lsm_hooks::on_capable(lsm_hooks::CapableContext::new(
69                    user_ns.as_ref(),
70                    ctx.posix_thread,
71                    CapSet::SYS_NICE,
72                ))?;
73            }
74        }
75        IoPrioClass::BestEffort | IoPrioClass::Idle => {}
76        IoPrioClass::None => {
77            if ioprio & IOPRIO_LEVEL_MASK != 0 {
78                return Err(errno!(EINVAL, "invalid I/O priority"));
79            }
80        }
81    }
82
83    Ok(())
84}
85
86// Linux stores the class in bits 13..=15 and the level in bits 0..=2.
87// Bits 3..=12 carry I/O hints and are not part of the level.
88const IOPRIO_CLASS_SHIFT: u32 = 13;
89const IOPRIO_CLASS_MASK: u32 = 0b111;
90const IOPRIO_LEVEL_MASK: u32 = 0b111;
91
92/// Valid I/O priority classes in the Linux ABI.
93#[repr(u32)]
94#[derive(Clone, Copy, Debug, TryFromInt)]
95enum IoPrioClass {
96    None = 0,
97    RealTime = 1,
98    BestEffort = 2,
99    Idle = 3,
100}
