49/// A `Path` is used to represent an exact location in the VFS tree.
50///
51/// Each `Path` corresponds to a node in the VFS tree, and a single node
52/// may have multiple `Path` instances referencing it due to mount operations.
53#[derive(Clone, Debug)]
54pub struct Path {
55    mount: Arc<Mount>,
56    dentry: Arc<Dentry>,
57}
58
59impl PartialEq for Path {
60    fn eq(&self, other: &Self) -> bool {
61        self.mount.id() == other.mount.id() && Arc::ptr_eq(&self.dentry, &other.dentry)
62    }
63}

74    /// Creates a new child `Path` in a file system.
75    ///
76    /// This API is for non-symbolic-link children.
77    /// Use [`Self::new_symlink_child`] to create a symbolic link, because a
78    /// symbolic link should be created atomically with its target.
79    pub(crate) fn new_child(&self, name: &str, type_: InodeType, mode: InodeMode) -> Result<Self> {
80        debug_assert_ne!(type_, InodeType::SymLink);
81
82        let dir_dentry = self.dentry.as_dir_dentry_or_err()?;
83        self.check_dir_entry_mutation()?;
84        let new_child_dentry = dir_dentry.create_child(name, || {
85            dir_dentry.inode().create(&dir_dentry, name, type_, mode)
86        })?;
87        Ok(Self::new(self.mount.clone(), new_child_dentry))
88    }

90    /// Creates a new symbolic-link child `Path` in a file system.
91    ///
92    /// Use this API instead of [`Self::new_child`] when creating symbolic
93    /// links so the link target is supplied at creation time.
94    pub(crate) fn new_symlink_child(
95        &self,
96        name: &str,
97        target: &str,
98        mode: InodeMode,
99    ) -> Result<Self> {
100        let dir_dentry = self.dentry.as_dir_dentry_or_err()?;
101        self.check_dir_entry_mutation()?;
102        let new_child_dentry = dir_dentry.create_child(name, || {
103            dir_dentry
104                .inode()
105                .create_symlink(&dir_dentry, name, target, mode)
106        })?;
107        Ok(Self::new(self.mount.clone(), new_child_dentry))
108    }

119    /// Creates a new `Path` to represent an unnamed temporary file.
120    ///
121    /// The returned inode has no directory entry and is invisible to `readdir`.
122    /// If created hard-linkable, it may later be given a name via a link
123    /// operation; otherwise it can never be linked.
124    pub(crate) fn create_tmpfile(
125        &self,
126        mode: InodeMode,
127        hard_linkability: HardLinkability,
128    ) -> Result<Self> {
129        let dir_dentry = self.dentry.as_dir_dentry_or_err()?;
130        self.check_dir_entry_mutation()?;
131        let tmp_inode = self
132            .inode()
133            .create_tmpfile(&self.dentry, mode, hard_linkability)?;
134        let tmp_dentry = Dentry::new_anonymous(tmp_inode, &dir_dentry);
135        Ok(Self::new(self.mount.clone(), tmp_dentry))
136    }

167    /// Opens the `Path` with the given `OpenArgs`.
168    ///
169    /// Returns an `InodeHandle` on success.
170    pub(crate) fn open(&self, open_args: OpenArgs) -> Result<InodeHandle> {
171        let inode = self.inode().as_ref();
172        let inode_type = inode.type_();
173        let creation_flags = &open_args.creation_flags;
174        let status_flags = &open_args.status_flags;
175
176        if creation_flags.contains(CreationFlags::O_CREAT)
177            && creation_flags.contains(CreationFlags::O_EXCL)
178        {
179            return_errno_with_message!(Errno::EEXIST, "the file already exists");
180        }
181        if creation_flags.contains(CreationFlags::O_CREAT) && inode_type == InodeType::Dir {
182            return_errno_with_message!(
183                Errno::EISDIR,
184                "O_CREAT is specified but the file is a directory"
185            );
186        }
187
188        if inode_type == InodeType::SymLink
189            && creation_flags.contains(CreationFlags::O_NOFOLLOW)
190            && !status_flags.contains(StatusFlags::O_PATH)
191        {
192            return_errno_with_message!(Errno::ELOOP, "the file is a symlink");
193        }
194        if creation_flags.contains(CreationFlags::O_DIRECTORY) && inode_type != InodeType::Dir {
195            return_errno_with_message!(
196                Errno::ENOTDIR,
197                "O_DIRECTORY is specified but the file is not a directory"
198            );
199        }
200
201        if inode_type.is_regular_file()
202            && creation_flags.contains(CreationFlags::O_TRUNC)
203            && !status_flags.contains(StatusFlags::O_PATH)
204        {
205            self.resize(0)?;
206        }
207
208        InodeHandle::new(self.clone(), open_args.access_mode, *status_flags)
209    }

211    /// Gets the parent `Path` within the same mount.
212    ///
213    /// This method returns the parent path within the same filesystem/mount.
214    /// It does NOT cross mount boundaries. If the current path is the root of a mount,
215    /// it will return `None`.
216    ///
217    /// For cross-filesystem parent lookup, use `effective_parent()` instead.
218    pub(super) fn parent_within_mount(&self) -> Option<Self> {
219        let parent = self.dentry.parent()?;
220        Some(Self::new(self.mount.clone(), parent))
221    }

223    /// Gets the top `Path` of the current.
224    ///
225    /// Used when different file systems are mounted on the same mount point.
226    ///
227    /// For example, first `mount /dev/sda1 /mnt` and then `mount /dev/sda2 /mnt`.
228    /// After the second mount is completed, the content of the first mount will be overridden.
229    /// We need to recursively obtain the top `Path`.
230    pub(crate) fn get_top_path(mut self) -> Self {
231        while self.dentry.is_mountpoint() {
232            if let Some(child_mount) = self.mount.get(&self.dentry) {
233                let inner = child_mount.root_dentry().clone();
234                self = Self::new(child_mount, inner);
235            } else {
236                break;
237            }
238        }
239
240        self
241    }

255    /// Checks if this path is reachable from the given `root` path.
256    ///
257    /// A path is considered reachable if it is the same as or a descendant
258    /// of the `root` path. The check traverses upwards from the current path,
259    /// crossing mount point boundaries as necessary, until it either finds
260    /// the `root` path or reaches the global root.
261    fn is_reachable_from(&self, root: &Path, _topology: &MountTopology) -> bool {
262        let mut owned;
263        let mut current = self;
264
265        loop {
266            if current.mount.id() == root.mount.id() {
267                return current.dentry.is_equal_or_descendant_of(&root.dentry);
268            }
269
270            let Some(parent_mount) = current.mount.parent().and_then(|mount| mount.upgrade())
271            else {
272                // `upgrade()` may fail if `current` is in a mount namespace that has been
273                // dropped. In this situation, `current` is not reachable from `root` because
274                // `root` holds strong reference counts for all its descendant mounts.
275                return false;
276            };
277
278            owned = Path::new(
279                parent_mount,
280                current
281                    .mount
282                    .mountpoint()
283                    .expect("Mounts with parents must have a mount point")
284                    .clone(),
285            );
286            current = &owned;
287        }
288    }

299    /// Checks whether this path is a searchable directory.
300    fn check_dir_search_permission(&self) -> Result<()> {
301        if self.type_() != InodeType::Dir {
302            return_errno_with_message!(Errno::ENOTDIR, "only directories are searchable");
303        }
304
305        self.inode()
306            .check_permission(Permission::MAY_EXEC | Permission::MAY_CHDIR)
307    }

309    /// Checks whether the path is on a writable mount and filesystem.
310    fn check_mount_writable(&self) -> Result<()> {
311        if self.mount.flags().contains(PerMountFlags::RDONLY)
312            || self.fs().flags().contains(FsFlags::RDONLY)
313        {
314            return_errno_with_message!(Errno::EROFS, "the mount or filesystem is read-only");
315        }
316
317        Ok(())
318    }

320    /// Checks whether a directory entry may be modified in this directory.
321    fn check_dir_entry_mutation(&self) -> Result<()> {
322        self.check_mount_writable()?;
323        self.inode()

325    }

327    /// Checks whether an inode may be used as the source of a hard link.
328    fn check_hardlink_source(&self) -> Result<()> {
329        let current_thread = current_thread!();
330        let Some(posix_thread) = current_thread.as_posix_thread() else {
331            return Ok(());
332        };
333
334        let metadata = self.metadata()?;
335        let fsuid = posix_thread.credentials().fsuid();
336
337        // The source inode owner can hardlink all they like.
338        if fsuid == metadata.uid {
339            return Ok(());
340        }
341
342        // Hardlinking to unreadable or unwritable sources is dangerous.
343        if self
344            .inode()
345            .check_permission(Permission::MAY_READ | Permission::MAY_WRITE)
346            .is_err()
347        {
348            return_errno_with_message!(
349                Errno::EPERM,
350                "the source is not permitted to be hard-linked"
351            );
352        }
353
354        // Non-owners may only hard-link regular files.
355        if self.type_() != InodeType::File {
356            return_errno_with_message!(
357                Errno::EPERM,
358                "the source is not permitted to be hard-linked"
359            );
360        }
361
362        // Hardlinking to setuid or executable setgid files requires CAP_FOWNER.
363        if (metadata.mode.has_set_uid()
364            || (metadata.mode.has_set_gid() && metadata.mode.is_group_executable()))
365            && lsm_hooks::on_capable(lsm_hooks::CapableContext::new(
366                UserNamespace::get_init_singleton().as_ref(),
367                posix_thread,
368                CapSet::FOWNER,
369            ))
370            .is_err()
371        {
372            return_errno_with_message!(
373                Errno::EPERM,
374                "the source is not permitted to be hard-linked"
375            );
376        }
377
378        Ok(())
379    }

389impl Path {
390    /// Mounts a filesystem at the current path.
391    ///
392    /// This method attaches a given filesystem to the VFS tree at the location
393    /// represented by `self`. The current path becomes the mountpoint for the new
394    /// filesystem.
395    ///
396    /// Returns the newly created child mount on success.
397    ///
398    /// # Errors
399    ///
400    /// Returns `ENOTDIR` if the path is not a directory.
401    /// Returns `EINVAL` if attempting to mount on root or if the path is not
402    /// in the current mount namespace.
403    pub(crate) fn mount(
404        &self,
405        fs_and_root: FsAndRoot,
406        flags: PerMountFlags,
407        source: Option<String>,
408        ctx: &Context,
409    ) -> Result<Arc<Mount>> {
410        if self.type_() != InodeType::Dir {
411            return_errno_with_message!(Errno::ENOTDIR, "the path is not a directory");
412        }
413
414        let fs_ref = ctx.thread_local.borrow_fs();
415        let path_resolver = fs_ref.resolver().read();
416
417        if path_resolver.root() == self {
418            return_errno_with_message!(Errno::EINVAL, "the root cannot be mounted on");
419        }
420
421        let current_ns_proxy = ctx.thread_local.borrow_ns_proxy();
422        let current_mnt_ns = current_ns_proxy.unwrap().mnt_ns();
423        if !current_mnt_ns.owns(&self.mount) {
424            return_errno_with_message!(Errno::EINVAL, "the path is not in this mount namespace");
425        }
426
427        let mut topology_guard = MountTopology::write_lock();
428        self.mount.do_mount(
429            fs_and_root,
430            flags,
431            &self.dentry,
432            source,
433            &mut topology_guard,
434        )
435    }

437    /// Unmounts the filesystem mounted at the current path.
438    ///
439    /// Returns the unmounted child mount on success.
440    ///
441    /// # Errors
442    ///
443    /// Returns `EINVAL` in the following cases:
444    /// - The current path is not a mount root.
445    /// - The mount of the current path is the root mount.
446    /// - The current path is not in the current mount namespace.
447    pub(crate) fn unmount(&self, ctx: &Context) -> Result<Arc<Mount>> {
448        if !self.is_mount_root() {
449            return_errno_with_message!(Errno::EINVAL, "the path is not a mount root");
450        }
451
452        let Some(_mountpoint) = self.mount.mountpoint() else {
453            return_errno_with_message!(Errno::EINVAL, "the root mount cannot be unmounted");
454        };
455
456        let current_ns_proxy = ctx.thread_local.borrow_ns_proxy();
457        let current_mnt_ns = current_ns_proxy.unwrap().mnt_ns();
458        if !current_mnt_ns.owns(&self.mount) {
459            return_errno_with_message!(Errno::EINVAL, "the path is not in this mount namespace");
460        }
461
462        self.mount.sync()?;
463
464        let mut topology_guard = MountTopology::write_lock();
465        let Some(mountpoint) = self.mount.mountpoint() else {
466            return_errno_with_message!(Errno::EINVAL, "the mount has been detached");
467        };
468        let parent_mount = self.mount.parent().unwrap().upgrade().unwrap();
469        let child_mount = parent_mount.do_unmount(&mountpoint, &mut topology_guard)?;
470
471        Ok(child_mount)
472    }

506    /// Creates a bind mount from the current path to the destination path.
507    ///
508    /// Creates a new mount tree that mirrors either the root mount (non-recursive)
509    /// or the entire mount subtree (recursive), and attaches it to the destination path.
510    ///
511    /// A recursive bind omits unbindable subtrees.
512    /// If the destination parent is shared,
513    /// the new tree is also attached below its propagation receivers.
514    ///
515    /// # Errors
516    ///
517    /// Returns `ENOTDIR` if one of the source and destination is a directory
518    /// and the other is not.
519    ///
520    /// Returns `EINVAL` if any of the following holds:
521    /// - The destination path is not in the current mount namespace.
522    /// - The source path is a mount namespace file that would create a namespace loop.
523    /// - The source mount is unbindable.
524    pub(crate) fn bind_mount_to(
525        &self,
526        dst_path: Self,
527        recursive: bool,
528        ctx: &Context,
529    ) -> Result<()> {
530        let can_bind = {
531            let src_is_dir = self.type_() == InodeType::Dir;
532            let dst_is_dir = dst_path.type_() == InodeType::Dir;
533            (src_is_dir && dst_is_dir) || (!src_is_dir && !dst_is_dir)
534        };
535        if !can_bind {
536            return_errno_with_message!(
537                Errno::ENOTDIR,
538                "the source and destination must both be directories or both be non-directories"
539            );
540        }
541
542        let current_ns_proxy = ctx.thread_local.borrow_ns_proxy();
543        let current_mnt_ns = current_ns_proxy.unwrap().mnt_ns();
544
545        // Linux does not require the source path to belong to the current
546        // mount namespace, but still rejects source mount namespace files that
547        // would form loops.
548        if !current_mnt_ns.owns(&dst_path.mount) {
549            return_errno_with_message!(
550                Errno::EINVAL,
551                "the destination path is not in this mount namespace"
552            );
553        }
554        if current_mnt_ns.would_form_mnt_ns_loop(&self.dentry) {
555            return_errno_with_message!(
556                Errno::EINVAL,
557                "the source is a mount namespace file that would create a namespace loop"
558            );
559        }
560
561        let mut topology_guard = MountTopology::write_lock();
562        let current_mnt_ns_weak = Arc::downgrade(current_mnt_ns);
563        let mut pending_changes = PendingPropagationChanges::default();
564        let new_mount = self.mount.clone_mount_tree_deferred(
565            &self.dentry,
566            &current_mnt_ns_weak,
567            recursive,
568            MountTreeCloneMode::Bind,
569            &mut pending_changes,
570            &topology_guard,
571        )?;
572        let attachment =
573            new_mount.prepare_mount_tree_attachment(dst_path, pending_changes, &topology_guard)?;
574        attachment.attach(&mut topology_guard);
575        Ok(())
576    }

578    /// Moves a mount tree from the current path to the destination path.
579    ///
580    /// If the destination parent is shared,
581    /// corresponding clones are also attached below its propagation receivers.
582    ///
583    /// # Errors
584    ///
585    /// Returns `EINVAL` in the following cases:
586    /// - The current path is not a mount root.
587    /// - The mount of the current path is the root mount.
588    /// - The source is a non-root mount in a detached mount tree.
589    /// - The destination is in a detached mount tree while the source is not.
590    /// - The source and destination are in the same detached mount tree.
591    /// - One of the source and destination is a directory and the other is not.
592    /// - The source mount's current parent is shared.
593    /// - The destination mount is shared and the source tree contains an
594    ///   unbindable mount.
595    ///
596    /// Returns `ELOOP` in the following cases:
597    /// - The destination path is inside the subtree being moved.
598    /// - The mount tree contains a mount namespace file that would create a namespace loop.
599    pub(crate) fn move_mount_to(&self, dst_path: Self, ctx: &Context) -> Result<()> {
600        if !self.is_mount_root() {
601            return_errno_with_message!(Errno::EINVAL, "the path is not a mount root");
602        };
603
604        let current_ns_proxy = ctx.thread_local.borrow_ns_proxy();
605        let current_mnt_ns = current_ns_proxy.unwrap().mnt_ns();
606        if self.mount.id() == current_mnt_ns.root().id() {
607            return_errno_with_message!(Errno::EINVAL, "the root mount can not be moved");
608        }
609        if !current_mnt_ns.owns(&self.mount) {
610            return_errno_with_message!(
611                Errno::EINVAL,
612                "the source path is not in this mount namespace"
613            );
614        }
615        if !current_mnt_ns.owns(&dst_path.mount) {
616            return_errno_with_message!(
617                Errno::EINVAL,
618                "the destination path is not in this mount namespace"
619            );
620        }
621
622        let mut topology_guard = MountTopology::write_lock();
623        let source_tree_root = mount_tree_root(self.mount_node(), &topology_guard);
624        let target_tree_root = mount_tree_root(dst_path.mount_node(), &topology_guard);
625        let current_tree_root = current_mnt_ns.root();
626        let source_is_detached = source_tree_root.id() != current_tree_root.id();
627        let target_is_detached = target_tree_root.id() != current_tree_root.id();
628
629        // Since Linux v6.15, a detached anonymous-namespace root can be
630        // moved onto either the current mount namespace or an acceptable
631        // anonymous namespace. Asterinas does not model anonymous mount
632        // namespaces yet, so compare detached mount tree roots instead.
633        // TODO: Once anonymous mount namespaces are modeled, determine
634        // detachedness from the mount namespace directly instead of inferring it
635        // from whether the mount tree root is the current namespace root.
636        if source_is_detached && self.mount.id() != source_tree_root.id() {
637            return_errno_with_message!(
638                Errno::EINVAL,
639                "only a detached mount tree root can be moved out of the detached tree"
640            );
641        }
642        if !source_is_detached && target_is_detached {
643            return_errno_with_message!(
644                Errno::EINVAL,
645                "the destination path is in a detached mount tree"
646            );
647        }
648        if source_is_detached && source_tree_root.id() == target_tree_root.id() {
649            return_errno_with_message!(
650                Errno::EINVAL,
651                "the source and destination are in the same detached mount tree"
652            );
653        }
654        let source_is_dir = self.type_() == InodeType::Dir;
655        let target_is_dir = dst_path.type_() == InodeType::Dir;
656        if source_is_dir != target_is_dir {
657            return_errno_with_message!(
658                Errno::EINVAL,
659                "the source and destination must both be directories or both be non-directories"
660            );
661        }
662
663        current_mnt_ns.check_no_mnt_ns_loop_in_tree(self.mount_node(), &topology_guard)?;
664        if dst_path
665            .mount_node()
666            .is_equal_or_descendant_of(self.mount_node(), &topology_guard)
667        {
668            // Reject moves that would place a mount beneath itself, because the mount tree
669            // must remain acyclic.
670            return_errno_with_message!(
671                Errno::ELOOP,
672                "the destination path is inside the mount subtree being moved"
673            );
674        }
675        if self
676            .mount
677            .parent()
678            .and_then(|parent| parent.upgrade())
679            .is_some_and(|parent| parent.is_shared())
680        {
681            return_errno_with_message!(
682                Errno::EINVAL,
683                "moving a mount under a shared parent is not supported"
684            );
685        }
686        if dst_path.mount_node().is_shared() {
687            self.mount.try_walk_tree(|mount| {
688                if mount.is_unbindable() {
689                    return_errno_with_message!(
690                        Errno::EINVAL,
691                        "an unbindable mount tree cannot be moved under a shared mount"
692                    );
693                }
694                Ok(())
695            })?;
696        }
697
698        let pending_changes = PendingPropagationChanges::default();
699        let attachment =
700            self.mount
701                .prepare_mount_tree_attachment(dst_path, pending_changes, &topology_guard)?;
702        attachment.attach(&mut topology_guard);
703
704        Ok(())
705    }
706
707    /// Sets the propagation policy of this path's mount.
708    ///
709    /// If `recursive` is `true`,
710    /// the same requested transition is applied independently
711    /// to every mount in the subtree.
712    ///
713    /// # Errors
714    ///
715    /// Returns `EINVAL` if this path is not a mount root
716    /// or its mount does not belong to the current mount namespace.
717    /// Returns `ENOMEM` if the transition requires a new peer group
718    /// and the peer-group ID pool is exhausted.
719    pub(crate) fn set_propagation(
720        &self,
721        prop: MountPropType,
722        recursive: bool,
723        ctx: &Context,
724    ) -> Result<()> {
725        if !self.is_mount_root() {
726            return_errno_with_message!(Errno::EINVAL, "the path is not a mount root");
727        };
728
729        let current_ns_proxy = ctx.thread_local.borrow_ns_proxy();
730        let current_mnt_ns = current_ns_proxy.unwrap().mnt_ns();
731        if !current_mnt_ns.owns(&self.mount) {
732            return_errno_with_message!(Errno::EINVAL, "the path is not in this mount namespace");
733        }
734
735        let mut topology_guard = MountTopology::write_lock();
736        let mut pending_changes = PendingPropagationChanges::default();
737        self.mount.set_propagation_deferred(
738            prop,
739            recursive,
740            &mut pending_changes,
741            &topology_guard,
742        )?;
743        pending_changes.commit(&mut topology_guard);
744        Ok(())
745    }
746}
747
748fn mount_tree_root(mount: &Arc<Mount>, _topology: &MountTopology) -> Arc<Mount> {
749    let mut root = mount.clone();
750    while let Some(parent) = root.parent().and_then(|parent| parent.upgrade()) {
751        root = parent;
752    }
753
754    root
755}

821    /// Changes the owner and/or group after checking mount writability and permissions.
822    ///
823    /// An ID of `None` leaves that ID unchanged.
824    /// A successful call on a non-directory clears the set-user-ID bit, even if
825    /// neither ID changes.
826    /// The set-group-ID bit is preserved only if the file is not group-executable
827    /// and the caller belongs to its group or has `CAP_FSETID`.
828    /// Clearing these bits requires the caller to own the file or have `CAP_FOWNER`.
829    pub(crate) fn chown(&self, uid: Option<Uid>, gid: Option<Gid>, ctx: &Context) -> Result<()> {
830        self.check_mount_writable()?;
831
832        // FIXME: Keep the ownership checks and owner, group, and mode updates in
833        // one inode critical section. `metadata()` returns a snapshot, so concurrent
834        // attribute changes can invalidate the checks before the updates run.
835        let metadata = self.metadata()?;
836        let credentials = ctx.posix_thread.credentials();
837        let user_ns = ctx.thread_local.borrow_user_ns();
838        let is_owner = credentials.fsuid() == metadata.uid;
839
840        // The owner may keep either ID or move the file into one of its groups.
841        // Other ownership changes require `CAP_CHOWN`.
842        let needs_cap_chown = uid.is_some_and(|uid| !is_owner || uid != metadata.uid)
843            || gid.is_some_and(|gid| {
844                !is_owner || (gid != metadata.gid && !credentials.has_group_for_fs_access(gid))
845            });
846        if needs_cap_chown {
847            lsm_hooks::on_capable(lsm_hooks::CapableContext::new(
848                user_ns.as_ref(),
849                ctx.posix_thread,
850                CapSet::CHOWN,
851            ))?;
852        }
853
854        // Directories retain their set-ID bits because set-group-ID controls
855        // group inheritance for newly created children.
856        let mut mode = metadata.mode;
857        if !matches!(self.type_(), InodeType::Dir) {
858            mode.remove(InodeMode::S_ISUID);
859            if mode.has_set_gid() {
860                let keeps_set_gid = !mode.is_group_executable()
861                    && (credentials.has_group_for_fs_access(metadata.gid)
862                        || lsm_hooks::on_capable(lsm_hooks::CapableContext::new(
863                            user_ns.as_ref(),
864                            ctx.posix_thread,
865                            CapSet::FSETID,
866                        ))
867                        .is_ok());
868                if !keeps_set_gid {
869                    mode.remove(InodeMode::S_ISGID);
870                }
871            }
872        }
873
874        if mode != metadata.mode {
875            // Clearing set-ID bits requires permission to change the file mode.
876            let may_change_mode = is_owner
877                || lsm_hooks::on_capable(lsm_hooks::CapableContext::new(
878                    user_ns.as_ref(),
879                    ctx.posix_thread,
880                    CapSet::FOWNER,
881                ))
882                .is_ok();
883            if !may_change_mode {
884                return_errno_with_message!(Errno::EPERM, "cannot change the file mode");
885            }
886            self.set_mode(mode)?;
887        }
888
889        if let Some(uid) = uid {
890            self.set_owner(uid)?;
891        }
892        if let Some(gid) = gid {
893            self.set_group(gid)?;
894        }
895
896        notify::on_attr_change(self);
897        Ok(())
898    }
899
900    /// Changes the mode after checking mount writability and the caller's permissions.
901    pub(crate) fn chmod(&self, mut mode: InodeMode, ctx: &Context) -> Result<()> {
902        self.check_mount_writable()?;
903
904        // FIXME: Keep the ownership checks and mode update in one inode critical
905        // section. `metadata()` returns a snapshot, so a concurrent `chown()` can
906        // change the owner or group before `set_mode()` runs.
907        let metadata = self.metadata()?;
908        let credentials = ctx.posix_thread.credentials();
909        let user_ns = ctx.thread_local.borrow_user_ns();
910
911        if credentials.fsuid() != metadata.uid {
912            lsm_hooks::on_capable(lsm_hooks::CapableContext::new(
913                user_ns.as_ref(),
914                ctx.posix_thread,
915                CapSet::FOWNER,
916            ))?;
917        }
918
919        // If the caller is not in the file's group and lacks `CAP_FSETID`,
920        // Linux silently clears the requested `S_ISGID` bit.
921        // Reference: <https://man7.org/linux/man-pages/man2/chmod.2.html>.
922        if mode.contains(InodeMode::S_ISGID)
923            && !credentials.has_group_for_fs_access(metadata.gid)
924            && lsm_hooks::on_capable(lsm_hooks::CapableContext::new(
925                user_ns.as_ref(),
926                ctx.posix_thread,
927                CapSet::FSETID,
928            ))
929            .is_err()
930        {
931            mode.remove(InodeMode::S_ISGID);
932        }
933
934        self.set_mode(mode)?;
935        notify::on_attr_change(self);
936        Ok(())
937    }

1062}

1067    let name_len = name_bytes.len();

1069        name_bytes[0] == DOT_BYTE
1070    } else if name_len == 2 {

1072    } else {
