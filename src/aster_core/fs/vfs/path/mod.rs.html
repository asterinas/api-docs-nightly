49/// A `Path` is used to represent an exact location in the VFS tree.
50///
51/// Each `Path` corresponds to a node in the VFS tree, and a single node
52/// may have multiple `Path` instances referencing it due to mount operations.
53#[derive(Clone, Debug)]
54pub struct Path {
55    mount: Arc<Mount>,
56    dentry: Arc<Dentry>,
57}
58
59impl PartialEq for Path {
60    fn eq(&self, other: &Self) -> bool {
61        self.mount.id() == other.mount.id() && Arc::ptr_eq(&self.dentry, &other.dentry)
62    }
63}

74    /// Creates a new child `Path` in a file system.
75    ///
76    /// This API is for non-symbolic-link children.
77    /// Use [`Self::new_symlink_child`] to create a symbolic link, because a
78    /// symbolic link should be created atomically with its target.
79    pub(crate) fn new_child(&self, name: &str, type_: InodeType, mode: InodeMode) -> Result<Self> {
80        debug_assert_ne!(type_, InodeType::SymLink);
81
82        let dir_dentry = self.dentry.as_dir_dentry_or_err()?;
83        self.check_dir_entry_mutation()?;
84        let new_child_dentry = dir_dentry.create_child(name, || {
85            dir_dentry.inode().create(&dir_dentry, name, type_, mode)
86        })?;
87        Ok(Self::new(self.mount.clone(), new_child_dentry))
88    }

91    ///
92    /// Use this API instead of [`Self::new_child`] when creating symbolic
93    /// links so the link target is supplied at creation time.

95        &self,
96        name: &str,
97        target: &str,

102        let new_child_dentry = dir_dentry.create_child(name, || {
103            dir_dentry

119    /// Creates a new `Path` to represent an unnamed temporary file.
120    ///
121    /// The returned inode has no directory entry and is invisible to `readdir`.
122    /// If created hard-linkable, it may later be given a name via a link
123    /// operation; otherwise it can never be linked.
124    pub(crate) fn create_tmpfile(
125        &self,
126        mode: InodeMode,
127        hard_linkability: HardLinkability,
128    ) -> Result<Self> {
129        let dir_dentry = self.dentry.as_dir_dentry_or_err()?;
130        self.check_dir_entry_mutation()?;
131        let tmp_inode = self
132            .inode()
133            .create_tmpfile(&self.dentry, mode, hard_linkability)?;
134        let tmp_dentry = Dentry::new_anonymous(tmp_inode, &dir_dentry);
135        Ok(Self::new(self.mount.clone(), tmp_dentry))
136    }

167    /// Opens the `Path` with the given `OpenArgs`.
168    ///
169    /// Returns an `InodeHandle` on success.
170    pub(crate) fn open(&self, open_args: OpenArgs) -> Result<InodeHandle> {
171        let inode = self.inode().as_ref();
172        let inode_type = inode.type_();
173        let creation_flags = &open_args.creation_flags;
174        let status_flags = &open_args.status_flags;
175
176        if creation_flags.contains(CreationFlags::O_CREAT)
177            && creation_flags.contains(CreationFlags::O_EXCL)
178        {
179            return_errno_with_message!(Errno::EEXIST, "the file already exists");
180        }
181        if creation_flags.contains(CreationFlags::O_CREAT) && inode_type == InodeType::Dir {
182            return_errno_with_message!(
183                Errno::EISDIR,
184                "O_CREAT is specified but the file is a directory"
185            );
186        }
187
188        if inode_type == InodeType::SymLink
189            && creation_flags.contains(CreationFlags::O_NOFOLLOW)
190            && !status_flags.contains(StatusFlags::O_PATH)
191        {
192            return_errno_with_message!(Errno::ELOOP, "the file is a symlink");
193        }
194        if creation_flags.contains(CreationFlags::O_DIRECTORY) && inode_type != InodeType::Dir {
195            return_errno_with_message!(
196                Errno::ENOTDIR,
197                "O_DIRECTORY is specified but the file is not a directory"
198            );
199        }
200
201        if inode_type.is_regular_file()
202            && creation_flags.contains(CreationFlags::O_TRUNC)
203            && !status_flags.contains(StatusFlags::O_PATH)
204        {
205            self.resize(0)?;
206        }
207
208        InodeHandle::new(self.clone(), open_args.access_mode, *status_flags)
209    }

211    /// Changes the mode after checking mount writability and the caller's permissions.
212    pub(crate) fn chmod(&self, mut mode: InodeMode, ctx: &Context) -> Result<()> {
213        self.check_mount_writable()?;
214
215        // FIXME: Keep the ownership checks and mode update in one inode critical
216        // section. `metadata()` returns a snapshot, so a concurrent `chown()` can
217        // change the owner or group before `set_mode()` runs.
218        let metadata = self.metadata()?;
219        let credentials = ctx.posix_thread.credentials();
220        let user_ns = ctx.thread_local.borrow_user_ns();
221
222        if credentials.fsuid() != metadata.uid {
223            lsm_hooks::on_capable(lsm_hooks::CapableContext::new(
224                user_ns.as_ref(),
225                ctx.posix_thread,
226                CapSet::FOWNER,
227            ))?;
228        }
229
230        // If the caller is not in the file's group and lacks `CAP_FSETID`,
231        // Linux silently clears the requested `S_ISGID` bit.
232        // Reference: <https://man7.org/linux/man-pages/man2/chmod.2.html>.
233        if mode.contains(InodeMode::S_ISGID)
234            && !credentials.has_group_for_fs_access(metadata.gid)
235            && lsm_hooks::on_capable(lsm_hooks::CapableContext::new(
236                user_ns.as_ref(),
237                ctx.posix_thread,
238                CapSet::FSETID,
239            ))
240            .is_err()
241        {
242            mode.remove(InodeMode::S_ISGID);
243        }
244
245        self.set_mode(mode)?;
246        notify::on_attr_change(self);
247        Ok(())
248    }

250    /// Changes the owner and/or group after checking mount writability and permissions.
251    ///
252    /// An ID of `None` leaves that ID unchanged.
253    /// A successful call on a non-directory clears the set-user-ID bit, even if
254    /// neither ID changes.
255    /// The set-group-ID bit is preserved only if the file is not group-executable
256    /// and the caller belongs to its group or has `CAP_FSETID`.
257    /// Clearing these bits requires the caller to own the file or have `CAP_FOWNER`.
258    pub(crate) fn chown(&self, uid: Option<Uid>, gid: Option<Gid>, ctx: &Context) -> Result<()> {
259        self.check_mount_writable()?;
260
261        // FIXME: Keep the ownership checks and owner, group, and mode updates in
262        // one inode critical section. `metadata()` returns a snapshot, so concurrent
263        // attribute changes can invalidate the checks before the updates run.
264        let metadata = self.metadata()?;
265        let credentials = ctx.posix_thread.credentials();
266        let user_ns = ctx.thread_local.borrow_user_ns();
267        let is_owner = credentials.fsuid() == metadata.uid;
268
269        // The owner may keep either ID or move the file into one of its groups.
270        // Other ownership changes require `CAP_CHOWN`.
271        let needs_cap_chown = uid.is_some_and(|uid| !is_owner || uid != metadata.uid)
272            || gid.is_some_and(|gid| {
273                !is_owner || (gid != metadata.gid && !credentials.has_group_for_fs_access(gid))
274            });
275        if needs_cap_chown {
276            lsm_hooks::on_capable(lsm_hooks::CapableContext::new(
277                user_ns.as_ref(),
278                ctx.posix_thread,
279                CapSet::CHOWN,
280            ))?;
281        }
282
283        // Directories retain their set-ID bits because set-group-ID controls
284        // group inheritance for newly created children.
285        let mut mode = metadata.mode;
286        if !matches!(self.type_(), InodeType::Dir) {
287            mode.remove(InodeMode::S_ISUID);
288            if mode.has_set_gid() {
289                let keeps_set_gid = !mode.is_group_executable()
290                    && (credentials.has_group_for_fs_access(metadata.gid)
291                        || lsm_hooks::on_capable(lsm_hooks::CapableContext::new(
292                            user_ns.as_ref(),
293                            ctx.posix_thread,
294                            CapSet::FSETID,
295                        ))
296                        .is_ok());
297                if !keeps_set_gid {
298                    mode.remove(InodeMode::S_ISGID);
299                }
300            }
301        }
302
303        if mode != metadata.mode {
304            // Clearing set-ID bits requires permission to change the file mode.
305            let may_change_mode = is_owner
306                || lsm_hooks::on_capable(lsm_hooks::CapableContext::new(
307                    user_ns.as_ref(),
308                    ctx.posix_thread,
309                    CapSet::FOWNER,
310                ))
311                .is_ok();
312            if !may_change_mode {
313                return_errno_with_message!(Errno::EPERM, "the file mode cannot be changed");
314            }
315            self.set_mode(mode)?;
316        }
317
318        if let Some(uid) = uid {
319            self.set_owner(uid)?;
320        }
321        if let Some(gid) = gid {
322            self.set_group(gid)?;
323        }
324        notify::on_attr_change(self);
325        Ok(())
326    }

340    /// Gets the top `Path` of the current.
341    ///
342    /// Used when different file systems are mounted on the same mount point.
343    ///
344    /// For example, first `mount /dev/sda1 /mnt` and then `mount /dev/sda2 /mnt`.
345    /// After the second mount is completed, the content of the first mount will be overridden.
346    /// We need to recursively obtain the top `Path`.
347    pub(crate) fn get_top_path(mut self) -> Self {
348        while self.dentry.is_mountpoint() {
349            if let Some(child_mount) = self.mount.get(&self.dentry) {
350                let inner = child_mount.root_dentry().clone();
351                self = Self::new(child_mount, inner);
352            } else {
353                break;
354            }
355        }
356
357        self
358    }

372    /// Checks if this path is reachable from the given `root` path.
373    ///
374    /// A path is considered reachable if it is the same as or a descendant
375    /// of the `root` path. The check traverses upwards from the current path,
376    /// crossing mount point boundaries as necessary, until it either finds
377    /// the `root` path or reaches the global root.
378    fn is_reachable_from(&self, root: &Path, _topology: &MountTopology) -> bool {
379        let mut owned;
380        let mut current = self;
381
382        loop {
383            if current.mount.id() == root.mount.id() {
384                return current.dentry.is_equal_or_descendant_of(&root.dentry);
385            }
386
387            let Some(parent_mount) = current.mount.parent().and_then(|mount| mount.upgrade())
388            else {
389                // `upgrade()` may fail if `current` is in a mount namespace that has been
390                // dropped. In this situation, `current` is not reachable from `root` because
391                // `root` holds strong reference counts for all its descendant mounts.
392                return false;
393            };
394
395            owned = Path::new(
396                parent_mount,
397                current
398                    .mount
399                    .mountpoint()
400                    .expect("Mounts with parents must have a mount point")
401                    .clone(),
402            );
403            current = &owned;
404        }
405    }

426    /// Checks whether the path is on a writable mount and filesystem.
427    fn check_mount_writable(&self) -> Result<()> {
428        if self.mount.flags().contains(PerMountFlags::RDONLY)
429            || self.fs().flags().contains(FsFlags::RDONLY)
430        {
431            return_errno_with_message!(Errno::EROFS, "the mount or filesystem is read-only");
432        }
433
434        Ok(())
435    }

444    /// Checks whether an inode may be used as the source of a hard link.
445    fn check_hardlink_source(&self) -> Result<()> {
446        let current_thread = current_thread!();
447        let Some(posix_thread) = current_thread.as_posix_thread() else {
448            return Ok(());
449        };
450
451        let metadata = self.metadata()?;
452        let fsuid = posix_thread.credentials().fsuid();
453
454        // The source inode owner can hardlink all they like.
455        if fsuid == metadata.uid {
456            return Ok(());
457        }
458
459        // Hardlinking to unreadable or unwritable sources is dangerous.
460        if self
461            .inode()
462            .check_permission(Permission::MAY_READ | Permission::MAY_WRITE)
463            .is_err()
464        {
465            return_errno_with_message!(
466                Errno::EPERM,
467                "the source is not permitted to be hard-linked"
468            );
469        }
470
471        // Non-owners may only hard-link regular files.
472        if self.type_() != InodeType::File {
473            return_errno_with_message!(
474                Errno::EPERM,
475                "the source is not permitted to be hard-linked"
476            );
477        }
478
479        // Hardlinking to setuid or executable setgid files requires CAP_FOWNER.
480        if (metadata.mode.has_set_uid()
481            || (metadata.mode.has_set_gid() && metadata.mode.is_group_executable()))
482            && lsm_hooks::on_capable(lsm_hooks::CapableContext::new(
483                UserNamespace::get_init_singleton().as_ref(),
484                posix_thread,
485                CapSet::FOWNER,
486            ))
487            .is_err()
488        {
489            return_errno_with_message!(
490                Errno::EPERM,
491                "the source is not permitted to be hard-linked"
492            );
493        }
494
495        Ok(())
496    }

506impl Path {
507    /// Mounts a filesystem at the current path.
508    ///
509    /// This method attaches a given filesystem to the VFS tree at the location
510    /// represented by `self`. The current path becomes the mountpoint for the new
511    /// filesystem.
512    ///
513    /// Returns the newly created child mount on success.
514    ///
515    /// # Errors
516    ///
517    /// Returns `ENOTDIR` if the path is not a directory.
518    /// Returns `EINVAL` if attempting to mount on root or if the path is not
519    /// in the current mount namespace.
520    pub(crate) fn mount(
521        &self,
522        fs_and_root: FsAndRoot,
523        flags: PerMountFlags,
524        source: Option<String>,
525        ctx: &Context,
526    ) -> Result<Arc<Mount>> {
527        if self.type_() != InodeType::Dir {
528            return_errno_with_message!(Errno::ENOTDIR, "the path is not a directory");
529        }
530
531        let fs_ref = ctx.thread_local.borrow_fs();
532        let path_resolver = fs_ref.resolver().read();
533
534        if path_resolver.root() == self {
535            return_errno_with_message!(Errno::EINVAL, "the root cannot be mounted on");
536        }
537
538        let current_ns_proxy = ctx.thread_local.borrow_ns_proxy();
539        let current_mnt_ns = current_ns_proxy.unwrap().mnt_ns();
540        if !current_mnt_ns.owns(&self.mount) {
541            return_errno_with_message!(Errno::EINVAL, "the path is not in this mount namespace");
542        }
543
544        let mut topology_guard = MountTopology::write_lock();
545        self.mount.do_mount(
546            fs_and_root,
547            flags,
548            &self.dentry,
549            source,
550            &mut topology_guard,
551        )
552    }
553
554    /// Unmounts the filesystem mounted at the current path.
555    ///
556    /// Returns the unmounted child mount on success.
557    ///
558    /// # Errors
559    ///
560    /// Returns `EINVAL` in the following cases:
561    /// - The current path is not a mount root.
562    /// - The mount of the current path is the root mount.
563    /// - The current path is not in the current mount namespace.
564    pub(crate) fn unmount(&self, ctx: &Context) -> Result<Arc<Mount>> {
565        if !self.is_mount_root() {
566            return_errno_with_message!(Errno::EINVAL, "the path is not a mount root");
567        }
568
569        let Some(_mountpoint) = self.mount.mountpoint() else {
570            return_errno_with_message!(Errno::EINVAL, "the root mount cannot be unmounted");
571        };
572
573        let current_ns_proxy = ctx.thread_local.borrow_ns_proxy();
574        let current_mnt_ns = current_ns_proxy.unwrap().mnt_ns();
575        if !current_mnt_ns.owns(&self.mount) {
576            return_errno_with_message!(Errno::EINVAL, "the path is not in this mount namespace");
577        }
578
579        self.mount.sync()?;
580
581        let mut topology_guard = MountTopology::write_lock();
582        let Some(mountpoint) = self.mount.mountpoint() else {
583            return_errno_with_message!(Errno::EINVAL, "the mount has been detached");
584        };
585        let parent_mount = self.mount.parent().unwrap().upgrade().unwrap();
586        let child_mount = parent_mount.do_unmount(&mountpoint, &mut topology_guard)?;
587
588        Ok(child_mount)
589    }

591    /// Remounts the filesystem with new `PerMountFlags` and optionally new `FsFlags`.
592    ///
593    /// If `fs_flags` is provided, it will update the flags of the mounted filesystem,
594    /// otherwise, only the flags of the current mount will be updated.
595    ///
596    /// # Errors
597    ///
598    /// Returns `EINVAL` in the following cases:
599    /// - The current path is not a mount root.
600    /// - The current path is not in the current mount namespace.
601    pub(crate) fn remount(
602        &self,
603        mount_flags: PerMountFlags,
604        fs_flags: Option<FsFlags>,
605        data: Option<&str>,
606        ctx: &Context,
607    ) -> Result<()> {
608        if !self.is_mount_root() {
609            return_errno_with_message!(Errno::EINVAL, "the path is not a mount root");
610        };
611
612        let current_ns_proxy = ctx.thread_local.borrow_ns_proxy();
613        let current_mnt_ns = current_ns_proxy.unwrap().mnt_ns();
614        if !current_mnt_ns.owns(&self.mount) {
615            return_errno_with_message!(Errno::EINVAL, "the path is not in this mount namespace");
616        }
617
618        let mut topology_guard = MountTopology::write_lock();
619        self.mount
620            .remount(mount_flags, fs_flags, data, ctx, &mut topology_guard)
621    }

623    /// Creates a bind mount from the current path to the destination path.
624    ///
625    /// Creates a new mount tree that mirrors either the root mount (non-recursive)
626    /// or the entire mount subtree (recursive), and attaches it to the destination path.
627    ///
628    /// A recursive bind omits unbindable subtrees.
629    /// If the destination parent is shared,
630    /// the new tree is also attached below its propagation receivers.
631    ///
632    /// # Errors
633    ///
634    /// Returns `ENOTDIR` if one of the source and destination is a directory
635    /// and the other is not.
636    ///
637    /// Returns `EINVAL` if any of the following holds:
638    /// - The destination path is not in the current mount namespace.
639    /// - The source path is a mount namespace file that would create a namespace loop.
640    /// - The source mount is unbindable.
641    pub(crate) fn bind_mount_to(
642        &self,
643        dst_path: Self,
644        recursive: bool,
645        ctx: &Context,
646    ) -> Result<()> {
647        let can_bind = {
648            let src_is_dir = self.type_() == InodeType::Dir;
649            let dst_is_dir = dst_path.type_() == InodeType::Dir;
650            (src_is_dir && dst_is_dir) || (!src_is_dir && !dst_is_dir)
651        };
652        if !can_bind {
653            return_errno_with_message!(
654                Errno::ENOTDIR,
655                "the source and destination must both be directories or both be non-directories"
656            );
657        }
658
659        let current_ns_proxy = ctx.thread_local.borrow_ns_proxy();
660        let current_mnt_ns = current_ns_proxy.unwrap().mnt_ns();
661
662        // Linux does not require the source path to belong to the current
663        // mount namespace, but still rejects source mount namespace files that
664        // would form loops.
665        if !current_mnt_ns.owns(&dst_path.mount) {
666            return_errno_with_message!(
667                Errno::EINVAL,
668                "the destination path is not in this mount namespace"
669            );
670        }
671        if current_mnt_ns.would_form_mnt_ns_loop(&self.dentry) {
672            return_errno_with_message!(
673                Errno::EINVAL,
674                "the source is a mount namespace file that would create a namespace loop"
675            );
676        }
677
678        let mut topology_guard = MountTopology::write_lock();
679        let current_mnt_ns_weak = Arc::downgrade(current_mnt_ns);
680        let mut pending_changes = PendingPropagationChanges::default();
681        let new_mount = self.mount.clone_mount_tree_deferred(
682            &self.dentry,
683            &current_mnt_ns_weak,
684            recursive,
685            MountTreeCloneMode::Bind,
686            &mut pending_changes,
687            &topology_guard,
688        )?;
689        let attachment =
690            new_mount.prepare_mount_tree_attachment(dst_path, pending_changes, &topology_guard)?;
691        attachment.attach(&mut topology_guard);
692        Ok(())
693    }
694
695    /// Moves a mount tree from the current path to the destination path.
696    ///
697    /// If the destination parent is shared,
698    /// corresponding clones are also attached below its propagation receivers.
699    ///
700    /// # Errors
701    ///
702    /// Returns `EINVAL` in the following cases:
703    /// - The current path is not a mount root.
704    /// - The mount of the current path is the root mount.
705    /// - The source is a non-root mount in a detached mount tree.
706    /// - The destination is in a detached mount tree while the source is not.
707    /// - The source and destination are in the same detached mount tree.
708    /// - One of the source and destination is a directory and the other is not.
709    /// - The source mount's current parent is shared.
710    /// - The destination mount is shared and the source tree contains an
711    ///   unbindable mount.
712    ///
713    /// Returns `ELOOP` in the following cases:
714    /// - The destination path is inside the subtree being moved.
715    /// - The mount tree contains a mount namespace file that would create a namespace loop.
716    pub(crate) fn move_mount_to(&self, dst_path: Self, ctx: &Context) -> Result<()> {
717        if !self.is_mount_root() {
718            return_errno_with_message!(Errno::EINVAL, "the path is not a mount root");
719        };
720
721        let current_ns_proxy = ctx.thread_local.borrow_ns_proxy();
722        let current_mnt_ns = current_ns_proxy.unwrap().mnt_ns();
723        if self.mount.id() == current_mnt_ns.root().id() {
724            return_errno_with_message!(Errno::EINVAL, "the root mount can not be moved");
725        }
726        if !current_mnt_ns.owns(&self.mount) {
727            return_errno_with_message!(
728                Errno::EINVAL,
729                "the source path is not in this mount namespace"
730            );
731        }
732        if !current_mnt_ns.owns(&dst_path.mount) {
733            return_errno_with_message!(
734                Errno::EINVAL,
735                "the destination path is not in this mount namespace"
736            );
737        }
738
739        let mut topology_guard = MountTopology::write_lock();
740        let source_tree_root = mount_tree_root(self.mount_node(), &topology_guard);
741        let target_tree_root = mount_tree_root(dst_path.mount_node(), &topology_guard);
742        let current_tree_root = current_mnt_ns.root();
743        let source_is_detached = source_tree_root.id() != current_tree_root.id();
744        let target_is_detached = target_tree_root.id() != current_tree_root.id();
745
746        // Since Linux v6.15, a detached anonymous-namespace root can be
747        // moved onto either the current mount namespace or an acceptable
748        // anonymous namespace. Asterinas does not model anonymous mount
749        // namespaces yet, so compare detached mount tree roots instead.
750        // TODO: Once anonymous mount namespaces are modeled, determine
751        // detachedness from the mount namespace directly instead of inferring it
752        // from whether the mount tree root is the current namespace root.
753        if source_is_detached && self.mount.id() != source_tree_root.id() {
754            return_errno_with_message!(
755                Errno::EINVAL,
756                "only a detached mount tree root can be moved out of the detached tree"
757            );
758        }
759        if !source_is_detached && target_is_detached {
760            return_errno_with_message!(
761                Errno::EINVAL,
762                "the destination path is in a detached mount tree"
763            );
764        }
765        if source_is_detached && source_tree_root.id() == target_tree_root.id() {
766            return_errno_with_message!(
767                Errno::EINVAL,
768                "the source and destination are in the same detached mount tree"
769            );
770        }
771        let source_is_dir = self.type_() == InodeType::Dir;
772        let target_is_dir = dst_path.type_() == InodeType::Dir;
773        if source_is_dir != target_is_dir {
774            return_errno_with_message!(
775                Errno::EINVAL,
776                "the source and destination must both be directories or both be non-directories"
777            );
778        }
779
780        current_mnt_ns.check_no_mnt_ns_loop_in_tree(self.mount_node(), &topology_guard)?;
781        if dst_path
782            .mount_node()
783            .is_equal_or_descendant_of(self.mount_node(), &topology_guard)
784        {
785            // Reject moves that would place a mount beneath itself, because the mount tree
786            // must remain acyclic.
787            return_errno_with_message!(
788                Errno::ELOOP,
789                "the destination path is inside the mount subtree being moved"
790            );
791        }
792        if self
793            .mount
794            .parent()
795            .and_then(|parent| parent.upgrade())
796            .is_some_and(|parent| parent.is_shared())
797        {
798            return_errno_with_message!(
799                Errno::EINVAL,
800                "moving a mount under a shared parent is not supported"
801            );
802        }
803        if dst_path.mount_node().is_shared() {
804            self.mount.try_walk_tree(|mount| {
805                if mount.is_unbindable() {
806                    return_errno_with_message!(
807                        Errno::EINVAL,
808                        "an unbindable mount tree cannot be moved under a shared mount"
809                    );
810                }
811                Ok(())
812            })?;
813        }
814
815        let pending_changes = PendingPropagationChanges::default();
816        let attachment =
817            self.mount
818                .prepare_mount_tree_attachment(dst_path, pending_changes, &topology_guard)?;
819        attachment.attach(&mut topology_guard);
820
821        Ok(())
822    }
823
824    /// Sets the propagation policy of this path's mount.
825    ///
826    /// If `recursive` is `true`,
827    /// the same requested transition is applied independently
828    /// to every mount in the subtree.
829    ///
830    /// # Errors
831    ///
832    /// Returns `EINVAL` if this path is not a mount root
833    /// or its mount does not belong to the current mount namespace.
834    /// Returns `ENOMEM` if the transition requires a new peer group
835    /// and the peer-group ID pool is exhausted.
836    pub(crate) fn set_propagation(
837        &self,
838        prop: MountPropType,
839        recursive: bool,
840        ctx: &Context,
841    ) -> Result<()> {
842        if !self.is_mount_root() {
843            return_errno_with_message!(Errno::EINVAL, "the path is not a mount root");
844        };
845
846        let current_ns_proxy = ctx.thread_local.borrow_ns_proxy();
847        let current_mnt_ns = current_ns_proxy.unwrap().mnt_ns();
848        if !current_mnt_ns.owns(&self.mount) {
849            return_errno_with_message!(Errno::EINVAL, "the path is not in this mount namespace");
850        }
851
852        let mut topology_guard = MountTopology::write_lock();
853        let mut pending_changes = PendingPropagationChanges::default();
854        self.mount.set_propagation_deferred(
855            prop,
856            recursive,
857            &mut pending_changes,
858            &topology_guard,
859        )?;
860        pending_changes.commit(&mut topology_guard);
861        Ok(())
862    }
863}
864
865fn mount_tree_root(mount: &Arc<Mount>, _topology: &MountTopology) -> Arc<Mount> {
866    let mut root = mount.clone();
867    while let Some(parent) = root.parent().and_then(|parent| parent.upgrade()) {
868        root = parent;
869    }
870
871    root
872}
