18/// Address-related attributes.
19///
20/// Reference: <https://elixir.bootlin.com/linux/v6.13/source/include/uapi/linux/if_addr.h#L26>.
21#[expect(non_camel_case_types)]
22#[expect(clippy::upper_case_acronyms)]
23#[repr(u16)]
24#[derive(Clone, Copy, Debug, Eq, PartialEq, TryFromInt)]
25pub(in route) enum AddrAttrClass {
26    UNSPEC = 0,
27    ADDRESS = 1,
28    LOCAL = 2,
29    LABEL = 3,
30    BROADCAST = 4,
31    ANYCAST = 5,
32    CACHEINFO = 6,
33    MULTICAST = 7,
34    FLAGS = 8,
35    RT_PRIORITY = 9,
36    TARGET_NETNSID = 10,
37    PROTO = 11,
38}
39
40/// The source protocol of an interface address.
41///
42/// Reference: <https://elixir.bootlin.com/linux/v7.1/source/include/uapi/linux/if_addr.h#L73>.
43#[repr(u8)]
44#[derive(Clone, Copy, Debug, Immutable, IntoBytes)]
45pub(crate) enum AddrProtocol {
46    /// An unspecified address source.
47    #[expect(dead_code)]
48    Unspecified = 0,
49    /// A loopback address configured by the kernel.
50    KernelLoopback = 1,
51    /// An address configured by the kernel from a Router Advertisement.
52    #[expect(dead_code)]
53    KernelRouterAdvertisement = 2,
54    /// A link-local address configured by the kernel.
55    #[expect(dead_code)]
56    KernelLinkLocal = 3,
57}
58
59#[derive(Debug)]
60pub(crate) enum AddrAttr {
61    Address(IpAddr),
62    Broadcast(Ipv4Addr),
63    Flags(AddrMessageFlags),
64    Label(CString),
65    Local(IpAddr),
66    Protocol(AddrProtocol),
67}
68
69impl AddrAttr {
70    pub(in route) fn class(&self) -> AddrAttrClass {
71        match self {
72            AddrAttr::Address(_) => AddrAttrClass::ADDRESS,
73            AddrAttr::Broadcast(_) => AddrAttrClass::BROADCAST,
74            AddrAttr::Flags(_) => AddrAttrClass::FLAGS,
75            AddrAttr::Label(_) => AddrAttrClass::LABEL,
76            AddrAttr::Local(_) => AddrAttrClass::LOCAL,
77            AddrAttr::Protocol(_) => AddrAttrClass::PROTO,
78        }
79    }
80}
81
82impl Attribute for AddrAttr {
83    fn type_(&self) -> u16 {
84        self.class() as u16
85    }
86
87    fn payload_as_bytes(&self) -> &[u8] {
88        match self {
89            AddrAttr::Address(address) => address.as_octets(),
90            AddrAttr::Broadcast(address) => address.as_octets(),
91            AddrAttr::Flags(flags) => flags.as_bytes(),
92            AddrAttr::Label(label) => label.as_bytes_with_nul(),
93            AddrAttr::Local(address) => address.as_octets(),
94            AddrAttr::Protocol(protocol) => protocol.as_bytes(),
95        }
96    }
97
98    fn read_from(header: &CAttrHeader, reader: &mut dyn MultiRead) -> Result<ContinueAttrRead<Self>>
99    where
100        Self: Sized,
101    {
102        let payload_len = header.payload_len();
103        reader.skip_some(payload_len);
104
105        // We currently support only GETADDR dump requests.
106        // Attributes are ignored unless strict checking is enabled,
107        // but we still need to inspect them to reject malformed or unsupported attributes
108        // when strict checking is enabled.
109        // TODO: Parse and validate the remaining attributes.
110
111        let Ok(class) = AddrAttrClass::try_from(header.type_()) else {
112            // Unknown attributes should be ignored.
113            // Reference: <https://docs.kernel.org/userspace-api/netlink/intro.html#unknown-attributes>.
114            return Ok(ContinueAttrRead::SkippedUnsupported);
115        };
116
117        match (class, payload_len) {
118            (AddrAttrClass::TARGET_NETNSID, 4) => {
119                // TODO: Parse the network namespace ID once we support network namespaces.
120                Ok(ContinueAttrRead::SkippedUnsupported)
121            }
122            (AddrAttrClass::TARGET_NETNSID, _) => {
123                warn!("address attribute `{:?}` contains invalid payload", class);
124                Ok(ContinueAttrRead::skipped_invalid(
125                    Errno::ERANGE,
126                    "the address attribute is invalid",
127                ))
128            }
129            _ => {
130                warn!("address attribute `{:?}` is not supported", class);
131                Ok(ContinueAttrRead::SkippedUnsupported)
132            }
133        }
134    }
135}
