26pub(super) fn do_get_addr(
27    request_segment: &AddrSegment,
28    strict_check: bool,
29) -> Result<Vec<RtnlSegment>> {
30    let dump_all = {
31        let flags = GetRequestFlags::from_bits_truncate(request_segment.header().flags);
32        flags.contains(GetRequestFlags::DUMP)
33    };
34    if !dump_all {
35        return_errno_with_message!(Errno::EOPNOTSUPP, "GETADDR only supports dump requests");
36    }
37
38    validate_getaddr_request(request_segment, strict_check)?;
39
40    let requested_family = request_segment.body().family;
41    // Linux filters address dumps by a nonzero interface index only in strict mode.
42    // References:
43    // - <https://elixir.bootlin.com/linux/v7.1/source/net/ipv4/devinet.c#L1950>
44    // - <https://elixir.bootlin.com/linux/v7.1/source/net/ipv6/addrconf.c#L5430>
45    let requested_index = strict_check
46        .then(|| NonZeroU32::new(request_segment.body().index))
47        .flatten();
48    let mut ifaces = iter_all_ifaces()
49        .filter(|iface| requested_index.is_none_or(|index| index.get() == iface.index()))
50        .peekable();
51    if requested_index.is_some() && ifaces.peek().is_none() {
52        return_errno_with_message!(Errno::ENODEV, "the interface does not exist");
53    }
54
55    let mut addr_segments: Vec<AddrSegment> = ifaces
56        .flat_map(|iface| iface_to_new_addrs(request_segment.header(), requested_family, iface))
57        .collect();
58
59    // Linux dumps IPv4 addresses before IPv6 addresses
60    // while preserving interface order within each family.
61    addr_segments.sort_by_key(|segment| segment.body().family);
62
63    let mut response_segments: Vec<RtnlSegment> = addr_segments
64        .into_iter()
65        .map(RtnlSegment::NewAddr)
66        .collect();
67
68    let response_mode = if requested_index.is_some() {
69        ResponseMode::FilteredDump
70    } else {
71        ResponseMode::Dump
72    };
73    util::finish_response(
74        request_segment.header(),
75        response_mode,
76        &mut response_segments,
77    );
78
79    Ok(response_segments)
80}

82fn validate_getaddr_request(request_segment: &AddrSegment, strict_check: bool) -> Result<()> {
83    if !strict_check {
84        return Ok(());
85    }
86
87    let body = request_segment.body();
88    // These fields are reserved in GETADDR dumps.
89    // Reference: <https://elixir.bootlin.com/linux/v7.1/source/net/ipv4/devinet.c#L1800>.
90    if body.prefix_len != 0 || body.flags != 0 || body.scope != 0 {
91        return_errno_with_message!(
92            Errno::EINVAL,
93            "the address request header contains nonzero prefix length, flags, or scope"
94        );
95    }
96
97    // Linux permits only `IFA_TARGET_NETNSID` in strict address dumps.
98    // Reference: <https://elixir.bootlin.com/linux/v7.1/source/net/ipv4/devinet.c#L1811>.
99    const ALLOWED: AddrAttrClass = AddrAttrClass::TARGET_NETNSID;
100    if request_segment
101        .attrs()
102        .iter()
103        .any(|attr| attr.class() != ALLOWED)
104    {
105        return_errno_with_message!(
106            Errno::EINVAL,
107            "the attribute is not allowed in an address dump request"
108        );
109    }
110    Ok(())
111}

113fn iface_to_new_addrs(
114    request_header: &CMsgSegHdr,
115    requested_family: u8,
116    iface: &Arc<Iface>,
117) -> impl IntoIterator<Item = AddrSegment> {
118    let mut addr_segments = [None, None];
119
120    // Linux dumps addresses for all families
121    // when the requested family is neither AF_INET nor AF_INET6.
122    let dump_ipv4 = requested_family != CSocketAddrFamily::AF_INET6 as u8;
123    let dump_ipv6 = requested_family != CSocketAddrFamily::AF_INET as u8;
124
125    if dump_ipv4 && let Some(cidr) = iface.ipv4_cidr() {
126        addr_segments[0] = Some(iface_to_new_addr(request_header, iface, IpCidr::Ipv4(cidr)));
127    };
128
129    if dump_ipv6 && let Some(cidr) = iface.ipv6_cidr() {
130        addr_segments[1] = Some(iface_to_new_addr(request_header, iface, IpCidr::Ipv6(cidr)));
131    }
132
133    addr_segments.into_iter().flatten()
134}

136fn iface_to_new_addr(
137    request_header: &CMsgSegHdr,
138    iface: &Arc<Iface>,
139    ip_cidr: IpCidr,
140) -> AddrSegment {
141    let (family, address, prefix_len, scope) = match ip_cidr {
142        IpCidr::Ipv4(ipv4_cidr) => {
143            let ipv4_addr = ipv4_cidr.address();
144            let scope = if ipv4_addr.is_loopback() {
145                RtScope::HOST
146            } else {
147                RtScope::UNIVERSE
148            };
149            (
150                CSocketAddrFamily::AF_INET,
151                ipv4_addr.into(),
152                ipv4_cidr.prefix_len(),
153                scope,
154            )
155        }
156        IpCidr::Ipv6(ipv6_cidr) => {
157            let ipv6_addr = ipv6_cidr.address();
158            let scope = if ipv6_addr.is_loopback() {
159                RtScope::HOST
160            } else if ipv6_addr.is_unicast_link_local() {
161                RtScope::LINK
162            } else {
163                RtScope::UNIVERSE
164            };
165            (
166                CSocketAddrFamily::AF_INET6,
167                ipv6_addr.into(),
168                ipv6_cidr.prefix_len(),
169                scope,
170            )
171        }
172    };
173
174    let header = CMsgSegHdr {
175        len: 0,
176        type_: CSegmentType::NEWADDR as _,
177        flags: SegHdrCommonFlags::empty().bits(),
178        seq: request_header.seq,
179        pid: request_header.pid,
180    };
181
182    let flags = AddrMessageFlags::PERMANENT;
183    let addr_message = CIfaddrMsg {
184        family: family as _,
185        prefix_len,
186        flags: flags.bits() as u8,
187        scope: scope as u8,
188        index: iface.index(),
189    };
190
191    let attrs = match address {
192        IpAddr::V4(_) => {
193            // Linux may report the following IPv4 address attributes, in order:
194            // `IFA_ADDRESS`, `IFA_LOCAL`, `IFA_BROADCAST`, `IFA_LABEL`, `IFA_PROTO`,
195            // `IFA_FLAGS`, `IFA_RT_PRIORITY`, and `IFA_CACHEINFO`.
196            // TODO: Support `IFA_PROTO`, `IFA_RT_PRIORITY`, and `IFA_CACHEINFO`.
197            // Reference: <https://elixir.bootlin.com/linux/v7.1/source/net/ipv4/devinet.c#L1759>.
198            let mut attrs = vec![
199                // On a point-to-point link, `IFA_ADDRESS` is the peer address.
200                // Otherwise, it equals `IFA_LOCAL`.
201                // Since `Iface` does not support peer addresses,
202                // report the local address here.
203                // Reference: <https://elixir.bootlin.com/linux/v7.1/source/include/uapi/linux/if_addr.h#L16>.
204                AddrAttr::Address(address),
205                AddrAttr::Local(address),
206            ];
207            if iface.flags().contains(InterfaceFlags::BROADCAST)
208                && let Some(broadcast_address) = iface.broadcast_addr()
209            {
210                attrs.push(AddrAttr::Broadcast(broadcast_address));
211            }
212            attrs.extend([
213                // Linux defaults `IFA_LABEL` to the interface name
214                // when no per-address label is configured.
215                // Since `Iface` does not support per-address labels,
216                // report the interface name.
217                // Reference: <https://elixir.bootlin.com/linux/v7.1/source/net/ipv4/devinet.c#L1765>.
218                AddrAttr::Label(iface.name().as_cstr().to_owned()),
219                AddrAttr::Flags(flags),
220            ]);
221            attrs
222        }
223        IpAddr::V6(_) => {
224            // Linux may report the following IPv6 address attributes, in order:
225            // `IFA_LOCAL`, `IFA_ADDRESS`, `IFA_RT_PRIORITY`, `IFA_CACHEINFO`,
226            // `IFA_FLAGS`, and `IFA_PROTO`.
227            // TODO: Support `IFA_RT_PRIORITY` and `IFA_CACHEINFO`.
228            // TODO: Support `IFA_PROTO` for non-loopback addresses.
229            // Reference: <https://elixir.bootlin.com/linux/v7.1/source/net/ipv6/addrconf.c#L5143>.
230            let mut attrs = vec![
231                // When a peer address is configured,
232                // `IFA_ADDRESS` is the peer address,
233                // and `IFA_LOCAL` is the local address.
234                // Otherwise, `IFA_ADDRESS` is the local address,
235                // and `IFA_LOCAL` is omitted.
236                // Since `Iface` does not support peer addresses,
237                // report only `IFA_ADDRESS` with the local address.
238                // Reference: <https://elixir.bootlin.com/linux/v7.1/source/net/ipv6/addrconf.c#L5189>.
239                AddrAttr::Address(address),
240                AddrAttr::Flags(flags),
241            ];
242            if address.is_loopback() {
243                // Linux marks an IPv6 loopback address with `IFAPROT_KERNEL_LO`.
244                // Reference: <https://elixir.bootlin.com/linux/v7.1/source/net/ipv6/addrconf.c#L3289>.
245                attrs.push(AddrAttr::Protocol(AddrProtocol::KernelLoopback));
246            }
247            attrs
248        }
249    };
250
251    AddrSegment::new(header, addr_message, attrs)
252}
