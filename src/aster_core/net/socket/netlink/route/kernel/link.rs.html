20/// The unspecified link-layer address.
21const UNSPECIFIED_LINK_ADDR: EthernetAddress = EthernetAddress([0; 6]);
22
23pub(super) fn do_get_link(
24    request_segment: &LinkSegment,
25    strict_check: bool,
26) -> Result<Vec<RtnlSegment>> {
27    validate_getlink_request(request_segment, strict_check)?;
28
29    let filter_by = FilterBy::from_request(request_segment)?;
30
31    let mut response_segments: Vec<RtnlSegment> = iter_all_ifaces()
32        // Filter to include only requested links.
33        .filter(|iface| match &filter_by {
34            FilterBy::Index(index) => *index == iface.index(),
35            FilterBy::Name(name) => *name == iface.name().as_cstr(),
36            FilterBy::Dump => true,
37        })
38        .map(|iface| iface_to_new_link(request_segment.header(), iface))
39        .map(RtnlSegment::NewLink)
40        .collect();
41
42    let response_mode = match filter_by {
43        FilterBy::Dump => ResponseMode::Dump,
44        FilterBy::Index(_) | FilterBy::Name(_) => ResponseMode::Single,
45    };
46
47    if response_mode == ResponseMode::Single && response_segments.is_empty() {
48        return_errno_with_message!(Errno::ENODEV, "no link found");
49    }
50
51    util::finish_response(
52        request_segment.header(),
53        response_mode,
54        &mut response_segments,
55    );
56
57    Ok(response_segments)
58}
59
60enum FilterBy<'a> {
61    Index(u32),
62    Name(&'a CStr),
63    Dump,
64}
65
66impl<'a> FilterBy<'a> {
67    fn from_request(request_segment: &'a LinkSegment) -> Result<Self> {
68        let dump_all = {
69            let flags = GetRequestFlags::from_bits_truncate(request_segment.header().flags);
70            flags.contains(GetRequestFlags::DUMP)
71        };
72        if dump_all {
73            return Ok(Self::Dump);
74        }
75
76        // `index` takes precedence over `required_name`.
77
78        let required_index = request_segment.body().index;
79        if required_index != 0 {
80            return Ok(Self::Index(required_index));
81        }
82
83        let required_name = request_segment.attrs().iter().find_map(|attr| {
84            if let LinkAttr::Name(name) = attr {
85                Some(name.as_cstr())
86            } else {
87                None
88            }
89        });
90        if let Some(required_name) = required_name {
91            return Ok(Self::Name(required_name));
92        }
93
94        return_errno_with_message!(
95            Errno::EINVAL,
96            "either interface name or index should be specified for non-dump mode"
97        );
98    }
99}

101fn validate_getlink_request(request_segment: &LinkSegment, strict_check: bool) -> Result<()> {
102    if !strict_check {
103        return Ok(());
104    }
105
106    let body = request_segment.body();
107    // GETLINK reserves these fields.
108    // Reference: <https://elixir.bootlin.com/linux/v7.1/source/net/core/rtnetlink.c#L4141>.
109    if body._pad != 0 || body.type_ != 0 || body.flags != 0 || body.change != 0 {
110        return_errno_with_message!(
111            Errno::EINVAL,
112            "the link request header contains nonzero padding, type, flags, or change"
113        );
114    }
115
116    // Linux accepts different attributes for link dumps and single-link queries
117    // when strict checking is enabled.
118    let flags = GetRequestFlags::from_bits_truncate(request_segment.header().flags);
119    let allowed = if flags.contains(GetRequestFlags::DUMP) {
120        if body.index != 0 {
121            return_errno_with_message!(
122                Errno::EINVAL,
123                "filtering by interface index is not supported for link dumps"
124            );
125        }
126        // Reference: <https://elixir.bootlin.com/linux/v7.1/source/net/core/rtnetlink.c#L2495>.
127        [
128            LinkAttrClass::MASTER,
129            LinkAttrClass::LINKINFO,
130            LinkAttrClass::EXT_MASK,
131            LinkAttrClass::IF_NETNSID,
132        ]
133        .as_slice()
134    } else {
135        // Reference: <https://elixir.bootlin.com/linux/v7.1/source/net/core/rtnetlink.c#L4152>.
136        [
137            LinkAttrClass::IFNAME,
138            LinkAttrClass::EXT_MASK,
139            LinkAttrClass::IF_NETNSID,
140        ]
141        .as_slice()
142    };
143    for attr in request_segment.attrs() {
144        if !allowed.contains(&attr.class()) {
145            return_errno_with_message!(
146                Errno::EINVAL,
147                "the attribute is not allowed in this link request"
148            );
149        }
150    }
151    Ok(())
152}

154fn iface_to_new_link(request_header: &CMsgSegHdr, iface: &Arc<Iface>) -> LinkSegment {
155    let header = CMsgSegHdr {
156        len: 0,
157        type_: CSegmentType::NEWLINK as _,
158        flags: SegHdrCommonFlags::empty().bits(),
159        seq: request_header.seq,
160        pid: request_header.pid,
161    };
162
163    let link_message = CIfinfoMsg {
164        family: CSocketAddrFamily::AF_UNSPEC as u8,
165        _pad: 0,
166        type_: iface.type_() as u16,
167        index: iface.index(),
168        flags: iface.flags().bits(),
169        change: 0,
170    };
171
172    // Linux may report dozens of attributes in a fixed order.
173    // See the reference below for the complete attribute list and ordering.
174    // TODO: Asterinas currently reports only a subset of these attributes.
175    // Reference: <https://elixir.bootlin.com/linux/v7.1/source/net/core/rtnetlink.c#L2050>.
176    let mut attrs = Vec::with_capacity(5);
177    attrs.extend([
178        LinkAttr::Name(*iface.name()),
179        LinkAttr::TxqLen(DEFAULT_TX_QUEUE_LEN),
180        LinkAttr::Mtu(iface.mtu() as u32),
181    ]);
182
183    let (link_addr, link_broadcast_addr) = match iface.ethernet_addr() {
184        Some(ethernet_addr) => (ethernet_addr, EthernetAddress::BROADCAST),
185        None => (UNSPECIFIED_LINK_ADDR, UNSPECIFIED_LINK_ADDR),
186    };
187    attrs.extend([
188        LinkAttr::Address(link_addr),
189        LinkAttr::Broadcast(link_broadcast_addr),
190    ]);
191
192    LinkSegment::new(header, link_message, attrs)
193}
