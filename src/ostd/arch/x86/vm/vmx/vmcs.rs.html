19/// A reference-counted VMCS.
20///
21/// Callers should disable preemption across loading and field access.
22pub(in crate::arch::vm) type Vmcs = Frame<VmcsRegionMeta>;
23
24pub(in crate::arch::vm) struct VmcsRegionMeta {
25    state: SpinLock<VmcsState>,
26}
27crate::impl_frame_meta_for!(VmcsRegionMeta);
28
29struct VmcsState {
30    active_cpu: Option<CpuId>,
31}
32
33impl Vmcs {
34    /// Allocates an inactive VMCS.
35    #[cfg_attr(not(ktest), expect(dead_code))]
36    pub(in crate::arch::vm) fn new(_vmx_guard: &VmxGuard) -> Result<Self> {
37        // SAFETY: `_vmx_guard` guarantees VMX support on every CPU.
38        let revision_id = unsafe { rdmsr(IA32_VMX_BASIC) } as u32 & 0x7fff_ffff;
39        let frame = FrameAllocOptions::new().alloc_frame_with(VmcsRegionMeta {
40            state: SpinLock::new(VmcsState { active_cpu: None }),
41        })?;
42        // SAFETY: The frame is exclusively owned and has never been activated.
43        unsafe { (paddr_to_vaddr(frame.paddr()) as *mut u32).write(revision_id) };
44
45        Ok(frame)
46    }
47
48    /// Makes this VMCS current, clearing it on its previous CPU if necessary.
49    ///
50    /// # Panics
51    ///
52    /// Local IRQs must be enabled.
53    #[cfg_attr(not(ktest), expect(dead_code))]
54    pub(in crate::arch::vm) fn load(&self, vmx_guard: &VmxGuard) -> Result<()> {
55        assert!(crate::arch::irq::is_local_enabled());
56        // SAFETY: `vmcs_state` is the locked state of `vmcs`.
57        unsafe { LocalVmcsState::do_activate(self, &mut self.meta().state.lock(), vmx_guard) }
58    }
59
60    /// Clears this VMCS on its active CPU.
61    ///
62    /// # Panics
63    ///
64    /// Local IRQs must be enabled.
65    #[cfg_attr(not(ktest), expect(dead_code))]
66    pub(in crate::arch::vm) fn deactivate(&self, vmx_guard: &VmxGuard) -> Result<()> {
67        assert!(crate::arch::irq::is_local_enabled());
68        // SAFETY: `vmcs_state` is the locked state of `vmcs`.
69        unsafe { LocalVmcsState::do_deactivate(self, &mut self.meta().state.lock(), vmx_guard) }
70    }
71
72    /// # Safety
73    ///
74    /// The VMCS must be current on this CPU.
75    #[cfg_attr(not(ktest), expect(dead_code))]
76    pub(in crate::arch::vm) unsafe fn read(
77        &self,
78        field: u32,
79        irq_guard: &DisabledLocalIrqGuard,
80    ) -> Result<usize> {
81        // SAFETY: The caller ensures safety.
82        unsafe { instructions::vmread(field, irq_guard) }
83    }
84
85    /// # Safety
86    ///
87    /// 1. The VMCS must be current on this CPU.
88    /// 2. The `field` and `value` must not violate the host kernel's state.
89    #[cfg_attr(not(ktest), expect(dead_code))]
90    pub(in crate::arch::vm) unsafe fn write(
91        &self,
92        field: u32,
93        value: usize,
94        irq_guard: &DisabledLocalIrqGuard,
95    ) -> Result<()> {
96        // SAFETY: The caller ensures safety.
97        unsafe { instructions::vmwrite(field, value, irq_guard) }
98    }
99}

101pub(super) struct LocalVmcsState {
102    current: Option<Paddr>,
103    active: BTreeMap<Paddr, Vmcs>,
104}
105
106cpu_local! {
107    static LOCAL_VMCS_STATE: RefCell<LocalVmcsState> =
108        RefCell::new(LocalVmcsState::new());
109    /// The VMCS regions queued for clearing on this CPU.
110    static PENDING_CLEAR: SpinLock<Vec<Paddr>, LocalIrqDisabled> =
111        SpinLock::new(Vec::new());
112}
113
114impl LocalVmcsState {
115    const fn new() -> Self {
116        Self {
117            current: None,
118            active: BTreeMap::new(),
119        }
120    }
121
122    /// # Safety
123    ///
124    /// `vmcs_state` must be the locked state of `vmcs`.
125    unsafe fn do_activate(
126        vmcs: &Vmcs,
127        vmcs_state: &mut VmcsState,
128        _vmx_guard: &VmxGuard,
129    ) -> Result<()> {
130        let preempt_guard = task::disable_preempt();
131        let previous_cpu = vmcs_state.active_cpu;
132        if let Some(active_cpu) = previous_cpu
133            && active_cpu != preempt_guard.current_cpu()
134        {
135            // SAFETY: `vmcs_state` is the locked state of `vmcs`.
136            unsafe { Self::do_deactivate(vmcs, vmcs_state, _vmx_guard)? };
137        }
138
139        let irq_guard = irq::disable_local();
140        let local = LOCAL_VMCS_STATE.get_with(&irq_guard);
141        let mut local = local.borrow_mut();
142        if local.current == Some(vmcs.paddr()) {
143            return Ok(());
144        }
145
146        let was_active = local.active.contains_key(&vmcs.paddr());
147        if !was_active {
148            if previous_cpu.is_none() {
149                // SAFETY:
150                // 1. `_vmx_guard` keeps this CPU in VMX root operation.
151                // 2. The VMCS region is initialized. `previous_cpu` is `None` means
152                //    this VMCS has not been active on any CPU before.
153                unsafe { instructions::vmclear(vmcs.paddr(), &irq_guard) }?;
154            }
155            local.active.insert(vmcs.paddr(), vmcs.clone());
156        }
157
158        // SAFETY:
159        // 1. `_vmx_guard` keeps this CPU in VMX root operation.
160        // 2. The region is initialized and cleared on any previous CPU.
161        if let Err(err) = unsafe { instructions::vmptrld(vmcs.paddr(), &irq_guard) } {
162            if !was_active {
163                local.active.remove(&vmcs.paddr());
164            }
165            return Err(err);
166        }
167        local.current = Some(vmcs.paddr());
168        vmcs_state.active_cpu = Some(preempt_guard.current_cpu());
169        Ok(())
170    }
171
172    /// # Safety
173    ///
174    /// `vmcs_state` must be the locked state of `vmcs`.
175    unsafe fn do_deactivate(
176        vmcs: &Vmcs,
177        vmcs_state: &mut VmcsState,
178        _vmx_guard: &VmxGuard,
179    ) -> Result<()> {
180        let Some(cpu) = vmcs_state.active_cpu else {
181            return Ok(());
182        };
183        let preempt_guard = task::disable_preempt();
184        if cpu == preempt_guard.current_cpu() {
185            let irq_guard = irq::disable_local();
186            // SAFETY:
187            // 1. This is the current CPU's state.
188            // 2  `_vmx_guard` keeps all CPUs in VMX root operation.
189            // 3. `vmcs.paddr()` is the physical address of the active VMCS region
190            //    on this CPU.
191            unsafe {
192                LOCAL_VMCS_STATE
193                    .get_with(&irq_guard)
194                    .borrow_mut()
195                    .clear(vmcs.paddr(), &irq_guard)?
196            };
197            vmcs_state.active_cpu = None;
198            return Ok(());
199        }
200
201        assert!(crate::arch::irq::is_local_enabled());
202        PENDING_CLEAR.get_on_cpu(cpu).lock().push(vmcs.paddr());
203
204        crate::smp::inter_processor_call(&CpuSet::from(cpu), || {
205            let irq_guard = irq::disable_local();
206            let pending_clear = PENDING_CLEAR.get_with(&irq_guard);
207            let mut pending_clear = pending_clear.lock();
208            let local = LOCAL_VMCS_STATE.get_with(&irq_guard);
209            let mut local = local.borrow_mut();
210            while let Some(paddr) = pending_clear.pop() {
211                // SAFETY:
212                // 1. This is the current CPU's state.
213                // 2. `_vmx_guard` keeps all CPUs in VMX root operation.
214                // 3. Each queued address identifies a retained VMCS region active
215                //    on this CPU.
216                if let Err(err) = unsafe { local.clear(paddr, &irq_guard) } {
217                    panic!("failed to clear remote VMCS: {:?}", err);
218                }
219            }
220        })
221        .wait();
222        vmcs_state.active_cpu = None;
223        Ok(())
224    }
225
226    /// Clears an active VMCS on the current CPU.
227    ///
228    /// # Safety
229    ///
230    /// 1. `self` must be the current CPU's VMCS state
231    /// 2. This CPU must remain in VMX root operation throughout the call.
232    /// 3. `paddr` must identify a valid, page-aligned, initialized VMCS region
233    ///    which is currently active on this CPU.
234    unsafe fn clear(&mut self, paddr: Paddr, irq_guard: &DisabledLocalIrqGuard) -> Result<()> {
235        // SAFETY: The caller ensures safety.
236        unsafe { instructions::vmclear(paddr, irq_guard) }?;
237
238        self.active.remove(&paddr);
239        if self.current == Some(paddr) {
240            self.current = None;
241        }
242        Ok(())
243    }
244
245    /// Clears every active VMCS on this CPU before VMX shutdown.
246    ///
247    /// # Safety
248    ///
249    /// 1. This CPU must remain in VMX root operation throughout the call.
250    /// 2. There is no concurrent VMCS activation and deactivation.
251    pub(super) unsafe fn deactivate_all(irq_guard: &DisabledLocalIrqGuard) -> Result<()> {
252        let local = LOCAL_VMCS_STATE.get_with(irq_guard);
253        // Deactivating without holding the VMCS's `state` lock is safe because
254        // the caller excludes concurrent VMCS activation and deactivation.
255        let active = {
256            let mut local = local.borrow_mut();
257            local.current = None;
258            core::mem::take(&mut local.active)
259        };
260        for (paddr, vmcs) in active {
261            // SAFETY:
262            // 1. The caller ensures this CPU remains in VMX root operation.
263            // 2. Each active VMCS region is valid, page-aligned, and initialized,
264            //    which is currently active on this CPU.
265            unsafe { instructions::vmclear(paddr, irq_guard) }?;
266            vmcs.meta().state.lock().active_cpu = None;
267        }
268        Ok(())
269    }
270}

272#[cfg(ktest)]
273mod test {
274    use x86::vmx::vmcs::guest::RIP;
275
276    use super::*;
277    use crate::arch::cpu::extension::{IsaExtensions, has_extensions};
278
279    #[ktest]
280    fn switch_and_reuse_vmcs() {
281        // Passes if VMX is not supported.
282        if !has_extensions(IsaExtensions::VMX) {
283            crate::early_print!(" [skipped: VMX unavailable]");
284            return;
285        }
286
287        let vmx = VmxGuard::acquire_vmx().expect("VMX is required");
288        let first = Vmcs::new(&vmx).unwrap();
289        let second = Vmcs::new(&vmx).unwrap();
290        let _preempt_guard = task::disable_preempt();
291
292        // Fields from the current VMCS can be read and written.
293        first.load(&vmx).unwrap();
294        unsafe { first.write(RIP, 0x1000, &irq::disable_local()).unwrap() };
295        second.load(&vmx).unwrap();
296        unsafe { second.write(RIP, 0x2000, &irq::disable_local()).unwrap() };
297        first.load(&vmx).unwrap();
298        assert_eq!(
299            unsafe { first.read(RIP, &irq::disable_local()).unwrap() },
300            0x1000
301        );
302
303        // VMX shutdown clears active VMCSs, which can be loaded again later.
304        drop(vmx);
305        assert!(first.meta().state.lock().active_cpu.is_none());
306        assert!(second.meta().state.lock().active_cpu.is_none());
307        let vmx = VmxGuard::acquire_vmx().unwrap();
308        first.load(&vmx).unwrap();
309        assert_eq!(
310            unsafe { first.read(RIP, &irq::disable_local()).unwrap() },
311            0x1000
312        );
313        second.load(&vmx).unwrap();
314        assert_eq!(
315            unsafe { second.read(RIP, &irq::disable_local()).unwrap() },
316            0x2000
317        );
318
319        // Clearing a non-current VMCS preserves the current one.
320        first.deactivate(&vmx).unwrap();
321        drop(first);
322        assert_eq!(
323            unsafe { second.read(RIP, &irq::disable_local()).unwrap() },
324            0x2000
325        );
326        second.deactivate(&vmx).unwrap();
327        drop(second);
328    }
