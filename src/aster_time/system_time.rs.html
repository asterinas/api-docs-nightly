3use core::{ops::Add, time::Duration};
4
5/// A system time, recorded as the duration since the Unix epoch.
6#[derive(Clone, Copy)]
7pub struct SystemTime(Duration);
8
9impl SystemTime {
10    /// The Unix epoch (1970-01-01 00:00:00).
11    pub const UNIX_EPOCH: Self = Self(Duration::ZERO);
12
13    /// Creates a new system time from the duration since the Unix epoch.
14    pub fn new(duration: Duration) -> Self {
15        Self(duration)
16    }
17
18    /// Returns the duration since the Unix epoch.
19    pub fn since_epoch(self) -> Duration {
20        self.0
21    }
22}
23
24impl Add<Duration> for SystemTime {
25    type Output = SystemTime;
26
27    fn add(self, rhs: Duration) -> Self::Output {
28        // We chose to use `saturating_add` to prevent panics if a buggy RTC reports an excessive
29        // duration near `Duration::MAX`.
30        //
31        // This means we'll lose some arithmetic properties, but as long as we only implement
32        // `Add`, it should be fine. For now, we don't implement other operations like `Sub`.
33        Self(self.0.saturating_add(rhs))
34    }
35}
