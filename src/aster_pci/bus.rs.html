3//! The PCI bus in the device model.
4//!
5//! [`PciBus`] uses [`PciCommonDevice`] as its device payload.
6//! Drivers registered on this bus must provide a slice of [`PciMatch`] rules for device matching.
7//! The bus exposes read-only `vendor`, `device`, `class`, and `revision` sysfs attributes on each [`PciDevice`].
8//! Its uevent environment includes `PCI_CLASS`, `PCI_ID`, and `PCI_SLOT_NAME`.
9
10use alloc::sync::Arc;
11
12use aster_device::{
13    bus::{Bus, BusDevice},
14    common::Attr,
15    uevent::UeventVars,
16};
17use ostd::Result;
18
19use crate::{capability::msix::CapabilityMsixData, common_device::PciCommonDevice};
20
21/// The PCI bus.
22pub struct PciBus;
23
24/// A device on the PCI bus.
25pub type PciDevice = BusDevice<PciBus>;
26
27/// PCI-specific operations on a device handle.
28pub trait PciDeviceExt {
29    /// Acquires necessary resources to build the MSI-X capability data, if the capability exists.
30    ///
31    /// Returns `Ok(None)` if the device has no MSI-X capability.
32    /// Dropping the returned capability disables its interrupts, releases its IRQ lines,
33    /// and allows it to be acquired again.
34    ///
35    /// # Errors
36    ///
37    /// - [`InvalidArgs`](ostd::Error::InvalidArgs) if the MSI-X table or pending bit array
38    ///   references a missing BAR or an I/O BAR instead of a memory BAR.
39    /// - [`AccessDenied`](ostd::Error::AccessDenied) if the required BAR I/O memory cannot
40    ///   be acquired, or the MSI-X capability is already acquired.
41    fn acquire_msix_capability(self: &Arc<Self>) -> Result<Option<CapabilityMsixData>>;
42}
43
44impl PciDeviceExt for PciDevice {
45    fn acquire_msix_capability(self: &Arc<Self>) -> Result<Option<CapabilityMsixData>> {
46        CapabilityMsixData::new(self)
47    }
48}
49
50/// Matching criteria for PCI drivers.
51#[derive(Clone, Copy, Debug)]
52pub enum PciMatch {
53    /// Matches a vendor and device IDs from `first` to `last`, both inclusive.
54    Device { vendor: u16, first: u16, last: u16 },
55    /// Matches a class, subclass, and programming interface.
56    Class {
57        class: u8,
58        subclass: u8,
59        prog_if: u8,
60    },
61}
62
63impl Bus for PciBus {
64    const NAME: &'static str = "pci";
65
66    type Device = PciCommonDevice;
67    type MatchData = [PciMatch];
68
69    fn matches(&self, dev: &PciCommonDevice, rules: &Self::MatchData) -> bool {
70        let id = dev.device_id();
71        rules.iter().any(|rule| match *rule {
72            PciMatch::Device {
73                vendor,
74                first,
75                last,
76            } => id.vendor_id == vendor && (first..=last).contains(&id.device_id),
77            PciMatch::Class {
78                class,
79                subclass,
80                prog_if,
81            } => id.class == class && id.subclass == subclass && id.prog_if == prog_if,
82        })
83    }
84
85    fn dev_attrs(&self) -> &'static [Attr<PciDevice>] {
86        const ATTRS: &[Attr<PciDevice>] = &[
87            Attr::ro("vendor", |dev, w| {
88                writeln!(w, "0x{:04x}", dev.device_id().vendor_id)?;
89                Ok(())
90            }),
91            Attr::ro("device", |dev, w| {
92                writeln!(w, "0x{:04x}", dev.device_id().device_id)?;
93                Ok(())
94            }),
95            Attr::ro("class", |dev, w| {
96                let id = dev.device_id();
97                writeln!(w, "0x{:02x}{:02x}{:02x}", id.class, id.subclass, id.prog_if)?;
98                Ok(())
99            }),
100            Attr::ro("revision", |dev, w| {
101                writeln!(w, "0x{:02x}", dev.device_id().revision_id)?;
102                Ok(())
103            }),
104        ];
105        ATTRS
106    }
107
108    fn append_uevent_vars(
109        &self,
110        dev: &PciDevice,
111        vars: &mut UeventVars,
112    ) -> aster_device::common::Result<()> {
113        // Match Linux's PCI uevent formatting for these variables:
114        // <https://elixir.bootlin.com/linux/v6.16/source/drivers/pci/pci-driver.c#L1549>.
115        // Omit `PCI_SUBSYS_ID` and `MODALIAS`
116        // because the PCI device model does not yet track subsystem IDs.
117
118        let id = dev.device_id();
119        let class =
120            (u32::from(id.class) << 16) | (u32::from(id.subclass) << 8) | u32::from(id.prog_if);
121        let location = dev.location();
122
123        vars.add("PCI_CLASS", format_args!("{class:04X}"))?;
124        vars.add(
125            "PCI_ID",
126            format_args!("{:04X}:{:04X}", id.vendor_id, id.device_id),
127        )?;
128        vars.add(
129            "PCI_SLOT_NAME",
130            format_args!(
131                "0000:{:02x}:{:02x}.{}",
132                location.bus, location.device, location.function
133            ),
134        )?;
135
136        Ok(())
137    }
138}
