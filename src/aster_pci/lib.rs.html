48// FIXME: Discover root buses from firmware instead of assuming bus 0 is the only root.
49// Until firmware root discovery is supported, use bus 0 as the root of the device tree.
50// This is true in most common cases. For example, QEMU documents this in
51// <https://github.com/qemu/qemu/blob/6883fb508cb7a93e2e7b44c0582bfd9fc2196d4d/docs/pcie.txt#L37>.
52const ROOT_BUS: u8 = 0;
53const MAX_BUSES_PER_SEGMENT: usize = 256;

55#[init_component]

57    init();
58    Ok(())

61/// The PCI bus instance.
62pub static PCI_BUS: Once<Arc<BusHandle<PciBus>>> = Once::new();
63
64fn init() {
65    let bus_handle = PCI_BUS.call_once(|| device_bus::register(PciBus).unwrap());
66    let Some(all_bus) = arch::init() else {
67        ostd::info!("no PCI bus was found");
68        return;
69    };
70    ostd::info!("initializing the PCI bus with bus numbers `{:?}`", all_bus);
71
72    if let Err(err) = enumerate(bus_handle, &all_bus) {
73        ostd::error!("cannot enumerate PCI devices: {:?}", err);
74    }
75}
76
77fn enumerate(bus_handle: &Arc<BusHandle<PciBus>>, range: &RangeInclusive<u8>) -> Result<()> {
78    let root = BareDevice::new_root(format!("pci0000:{ROOT_BUS:02x}"));
79    aster_device::add_device(&root)?;
80    scan_bus(
81        bus_handle,
82        range,
83        ROOT_BUS,
84        root,
85        &mut [false; MAX_BUSES_PER_SEGMENT],
86    )
87}
88
89fn scan_bus(
90    bus_handle: &Arc<BusHandle<PciBus>>,
91    range: &RangeInclusive<u8>,
92    bus: u8,
93    parent: Arc<dyn AnyDevice>,
94    visited: &mut [bool; MAX_BUSES_PER_SEGMENT],
95) -> Result<()> {
96    if !range.contains(&bus) {
97        return Err(Error::InvalidTopology);
98    }
99
100    // A bus may be reachable through multiple bridges, as on the i450NX chipset.
101    // Skip buses already scanned to avoid duplicate devices and recursion cycles.
102    // Reference: <https://elixir.bootlin.com/linux/v6.17/source/drivers/pci/probe.c#L1400-L1405>.
103    if visited[bus as usize] {
104        return Ok(());
105    }
106    visited[bus as usize] = true;
107
108    for device in PciDeviceLocation::MIN_DEVICE..=PciDeviceLocation::MAX_DEVICE {
109        let mut location = PciDeviceLocation {
110            bus,
111            device,
112            function: PciDeviceLocation::MIN_FUNCTION,
113        };
114        let Some(common_device) = PciCommonDevice::new(location) else {
115            continue;
116        };
117        let has_multi_function = common_device.has_multi_funcs();
118        register_function(bus_handle, range, common_device, &parent, visited)?;
119
120        if !has_multi_function {
121            continue;
122        }
123        for function in (PciDeviceLocation::MIN_FUNCTION + 1)..=PciDeviceLocation::MAX_FUNCTION {
124            location.function = function;
125            if let Some(common_device) = PciCommonDevice::new(location) {
126                register_function(bus_handle, range, common_device, &parent, visited)?;
127            }
128        }
129    }
130    Ok(())
131}
132
133fn register_function(
134    bus_handle: &Arc<BusHandle<PciBus>>,
135    range: &RangeInclusive<u8>,
136    common_device: PciCommonDevice,
137    parent: &Arc<dyn AnyDevice>,
138    visited: &mut [bool; MAX_BUSES_PER_SEGMENT],
139) -> Result<()> {
140    let device_type = common_device.device_type();
141    let location = *common_device.location();
142    let name = format!(
143        "0000:{:02x}:{:02x}.{}",
144        location.bus, location.device, location.function
145    );
146    let device = PciDevice::builder(bus_handle, name, common_device)
147        .parent(parent.clone())
148        .build();
149    aster_device::add_device(&device)?;
150
151    const UNCONFIGURED_SECONDARY_BUS: u8 = 0;
152    if let PciDeviceType::PciToPciBridge(_, secondary, _) = device_type
153        && secondary != UNCONFIGURED_SECONDARY_BUS
154    {
155        // An invalid bridge edge must not stop enumeration of unrelated devices.
156        match scan_bus(bus_handle, range, secondary, device, visited) {
157            Err(Error::InvalidTopology) => ostd::warn!(
158                "skipping bridge {:02x}:{:02x}.{} secondary bus {:02x}: outside accessible range",
159                location.bus,
160                location.device,
161                location.function,
162                secondary
163            ),
164            result => result?,
165        }
166    }
167    Ok(())
168}

170#[derive(Debug)]
171enum Error {
172    /// A bus number is outside the accessible range.
173    InvalidTopology,
174    /// The device model failed to register a PCI function or root device.
175    DeviceModel(#[expect(dead_code)] aster_device::common::Error),
176}
177
178type Result<T, E = Error> = core::result::Result<T, E>;
179
180impl From<aster_device::common::Error> for Error {
181    fn from(error: aster_device::common::Error) -> Self {
182        Self::DeviceModel(error)
183    }
