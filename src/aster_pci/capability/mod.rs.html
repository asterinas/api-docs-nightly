22/// Raw PCI Capabilities.
23#[derive(Debug, Default)]
24pub(super) struct RawCapabilities {
25    msix: Option<RawCapabilityMsix>,
26    vndr: Vec<RawCapabilityVndr>,
27}
28
29/// PCI capability types.
30#[repr(u8)]
31#[derive(Clone, Copy, Debug, TryFromInt)]
32enum CapabilityType {
33    /// Power Management
34    Pm = 0x01,
35    /// Accelerated Graphics Part
36    Agp = 0x02,
37    /// Vital Product Data
38    Vpd = 0x03,
39    /// Slot Identification
40    SlotId = 0x04,
41    /// Message Signalled Interrupts
42    Msi = 0x05,
43    /// CompactPCI HotSwap
44    Chswp = 0x06,
45    /// PCI-X
46    PciX = 0x07,
47    /// HyperTransport
48    Hp = 0x08,
49    /// Vendor-Specific
50    Vndr = 0x09,
51    /// Debug port
52    Dbg = 0x0A,
53    /// CompactPCI Central Resource Control
54    Ccrc = 0x0B,
55    /// PCI Standard Hot-Plug Controller
56    Shpc = 0x0C,
57    /// Bridge subsystem vendor/device ID
58    Ssvid = 0x0D,
59    /// AGP Target PCI-PCI bridge
60    Agp3 = 0x0E,
61    /// Secure Device
62    Secdev = 0x0F,
63    /// PCI Express
64    Exp = 0x10,
65    /// MSI-X
66    Msix = 0x11,
67    /// SATA Data/Index Conf
68    Sata = 0x12,
69    /// PCI Advanced Features
70    Af = 0x13,
71    /// Enhanced Allocation
72    Ea = 0x14,
73}
74
75impl RawCapabilities {
76    /// The top of the capability position.
77    const CAPABILITY_TOP: u16 = 0xFC;
78
79    /// Parses the capabilities of the PCI device.
80    pub(super) fn parse(dev: &PciCommonDevice) -> Self {
81        if !dev.read_status().contains(Status::CAPABILITIES_LIST) {
82            return Self::default();
83        }
84
85        // The offset of the first capability pointer is the same for PCI general devices and PCI
86        // bridge devices.
87        const CAP_OFFSET: u16 = PciGeneralDeviceCfgOffset::CapabilitiesPointer as u16;
88        let mut cap_ptr =
89            (dev.location().read8(CAP_OFFSET) as u16).align_down(align_of::<u32>() as _);
90        let mut cap_ptr_vec = Vec::new();
91
92        // Read all capability pointers so that it is easy for us to get the length of each
93        // capability.
94        while cap_ptr > 0 {
95            cap_ptr_vec.push(cap_ptr);
96            cap_ptr = (dev.location().read8(cap_ptr + 1) as u16).align_down(align_of::<u32>() as _);
97        }
98        cap_ptr_vec.sort();
99
100        // Push the top position so that we can calculate the length of the last capability.
101        cap_ptr_vec.push(Self::CAPABILITY_TOP);
102
103        let mut caps = Self::default();
104
105        let length = cap_ptr_vec.len();
106        for i in 0..length - 1 {
107            let cap_ptr = cap_ptr_vec[i];
108            let next_ptr = cap_ptr_vec[i + 1];
109            let raw_cap_type = dev.location().read8(cap_ptr);
110
111            let Ok(cap_type) = CapabilityType::try_from(raw_cap_type) else {
112                continue;
113            };
114            match cap_type {
115                CapabilityType::Msix => {
116                    // "More than one MSI-X Capability structure per Function is prohibited."
117                    if caps.msix.is_some() {
118                        ostd::warn!(
119                            "superfluous MSI-X Capability structures at {:?} are ignored",
120                            dev.location()
121                        );
122                        continue;
123                    }
124                    caps.msix = Some(RawCapabilityMsix::parse(dev, cap_ptr));
125                }
126                CapabilityType::Vndr => {
127                    caps.vndr
128                        .push(RawCapabilityVndr::new(cap_ptr, next_ptr - cap_ptr));
129                }
130                _ => {}
131            }
132        }
133
134        caps
135    }
136
137    /// Returns the raw MSI-X capability, if present.
138    pub(super) fn msix(&self) -> Option<&RawCapabilityMsix> {
139        self.msix.as_ref()
140    }
141
142    /// Iterates over [`CapabilityVndrData`] instances.
143    pub(super) fn iter_vndr_data<'a>(
144        &'a self,
145        device: &'a PciCommonDevice,
146    ) -> impl Iterator<Item = CapabilityVndrData<'a>> {
147        self.vndr
148            .iter()
149            .map(move |raw_vndr| CapabilityVndrData::new(device, raw_vndr))
150    }
151}
