9impl DrmFile {
10    pub(super) fn dispatch_ioctl(&self, raw_ioctl: RawIoctl) -> Result<i32> {
11        dispatch_ioctl!(match raw_ioctl {
12            // General ioctl cmds.
13            cmd @ DrmIoctlVersion => {
14                self.check_ioctl_requirements(DrmIoctlAccess::RENDER_ALLOW, DrmFeatures::empty())?;
15                self.drm_get_version(cmd)
16            }
17            cmd @ DrmIoctlGetUnique => {
18                self.check_ioctl_requirements(DrmIoctlAccess::empty(), DrmFeatures::empty())?;
19                self.drm_get_unique(cmd)
20            }
21            cmd @ DrmIoctlGetMagic => {
22                self.check_ioctl_requirements(DrmIoctlAccess::empty(), DrmFeatures::empty())?;
23                self.drm_get_magic(cmd)
24            }
25            cmd @ DrmIoctlGetCap => {
26                self.check_ioctl_requirements(DrmIoctlAccess::RENDER_ALLOW, DrmFeatures::empty())?;
27                self.drm_get_cap(cmd)
28            }
29            cmd @ DrmIoctlSetClientCap => {
30                self.check_ioctl_requirements(DrmIoctlAccess::empty(), DrmFeatures::MODESET)?;
31                self.drm_set_client_cap(cmd)
32            }
33            cmd @ DrmIoctlAuthMagic => {
34                self.check_ioctl_requirements(DrmIoctlAccess::MASTER, DrmFeatures::empty())?;
35                self.drm_auth_magic(cmd)
36            }
37            cmd @ DrmIoctlSetMaster => {
38                self.check_ioctl_requirements(DrmIoctlAccess::empty(), DrmFeatures::empty())?;
39                self.drm_set_master(cmd)
40            }
41            cmd @ DrmIoctlDropMaster => {
42                self.check_ioctl_requirements(DrmIoctlAccess::empty(), DrmFeatures::empty())?;
43                self.drm_drop_master(cmd)
44            }
45            _ => {
46                ostd::warn!(
47                    "unknown ioctl minor={:?} cmd={:#x}",
48                    self.minor_type(),
49                    raw_ioctl.cmd()
50                );
51                return_errno_with_message!(Errno::ENOTTY, "the DRM ioctl command is unknown")
52            }
53        })
54    }
55
56    fn check_ioctl_requirements(
57        &self,
58        required_flags: DrmIoctlAccess,
59        required_features: DrmFeatures,
60    ) -> Result<()> {
61        if !self.has_features(required_features) {
62            return_errno_with_message!(
63                Errno::EOPNOTSUPP,
64                "the DRM device lacks a feature required by the ioctl"
65            );
66        }
67
68        match self.minor_type() {
69            DrmMinorType::Primary => {
70                if required_flags.contains(DrmIoctlAccess::AUTH) && !self.is_authenticated() {
71                    return_errno_with_message!(
72                        Errno::EACCES,
73                        "the DRM ioctl requires an authenticated primary client"
74                    );
75                }
76            }
77            DrmMinorType::Render => {
78                if !required_flags.contains(DrmIoctlAccess::RENDER_ALLOW) {
79                    return_errno_with_message!(
80                        Errno::EACCES,
81                        "the DRM ioctl is not allowed on a render node"
82                    );
83                }
84            }
85        }
86
87        if required_flags.contains(DrmIoctlAccess::ROOT_ONLY) && !has_current_sys_admin() {
88            return_errno_with_message!(Errno::EACCES, "the DRM ioctl requires CAP_SYS_ADMIN");
89        }
90        if required_flags.contains(DrmIoctlAccess::MASTER) && !self.is_master() {
91            return_errno_with_message!(Errno::EACCES, "the DRM client is not the current master");
92        }
93
94        Ok(())
95    }
96}
97
98bitflags::bitflags! {
99    struct DrmIoctlAccess: u32 {
100        /// Requires the client to be authenticated or the current master.
101        const AUTH = 1 << 0;
102        /// Requires the current file to be the DRM master.
103        const MASTER = 1 << 1;
104        /// Requires the caller to have `CAP_SYS_ADMIN`.
105        const ROOT_ONLY = 1 << 2;
106        /// Allows the ioctl on a render node.
107        const RENDER_ALLOW = 1 << 3;
108    }
109}

111mod ioctl_defs {
112    use aster_core::{
113        ioc,
114        util::ioctl::{InData, InOutData, NoData, OutData},
115    };
116
117    use super::general::{DrmAuth, DrmGetCap, DrmSetClientCap, DrmUnique, DrmVersion};
118
119    pub(super) type DrmIoctlVersion =
120        ioc!(DRM_IOCTL_VERSION, b'd', 0x00, InOutData<DrmVersion>);
121    pub(super) type DrmIoctlGetUnique =
122        ioc!(DRM_IOCTL_GET_UNIQUE, b'd', 0x01, InOutData<DrmUnique>);
123    pub(super) type DrmIoctlGetMagic =
124        ioc!(DRM_IOCTL_GET_MAGIC, b'd', 0x02, OutData<DrmAuth>);
125    pub(super) type DrmIoctlGetCap =
126        ioc!(DRM_IOCTL_GET_CAP, b'd', 0x0c, InOutData<DrmGetCap>);
127    pub(super) type DrmIoctlSetClientCap =
128        ioc!(DRM_IOCTL_SET_CLIENT_CAP, b'd', 0x0d, InData<DrmSetClientCap>);
129    pub(super) type DrmIoctlAuthMagic =
130        ioc!(DRM_IOCTL_AUTH_MAGIC, b'd', 0x11, InData<DrmAuth>);
131    pub(super) type DrmIoctlSetMaster = ioc!(DRM_IOCTL_SET_MASTER, b'd', 0x1e, NoData);
132    pub(super) type DrmIoctlDropMaster = ioc!(DRM_IOCTL_DROP_MASTER, b'd', 0x1f, NoData);
