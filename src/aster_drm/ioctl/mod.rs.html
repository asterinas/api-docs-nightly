11impl DrmFile {
12    pub(super) fn dispatch_ioctl(&self, raw_ioctl: RawIoctl) -> Result<i32> {
13        dispatch_ioctl!(match raw_ioctl {
14            // General ioctl cmds.
15            cmd @ DrmIoctlVersion => {
16                self.check_ioctl_access(DrmIoctlAccess::RENDER_ALLOW)?;
17                self.drm_get_version(cmd)
18            }
19            cmd @ DrmIoctlGetUnique => {
20                self.check_ioctl_access(DrmIoctlAccess::empty())?;
21                self.drm_get_unique(cmd)
22            }
23            cmd @ DrmIoctlGetMagic => {
24                self.check_ioctl_access(DrmIoctlAccess::empty())?;
25                self.drm_get_magic(cmd)
26            }
27            cmd @ DrmIoctlGetCap => {
28                self.check_ioctl_access(DrmIoctlAccess::RENDER_ALLOW)?;
29                self.drm_get_cap(cmd)
30            }
31            cmd @ DrmIoctlSetClientCap => {
32                self.check_ioctl_access(DrmIoctlAccess::empty())?;
33                self.drm_set_client_cap(cmd)
34            }
35            cmd @ DrmIoctlAuthMagic => {
36                self.check_ioctl_access(DrmIoctlAccess::MASTER)?;
37                self.drm_auth_magic(cmd)
38            }
39            cmd @ DrmIoctlSetMaster => {
40                self.check_ioctl_access(DrmIoctlAccess::empty())?;
41                self.drm_set_master(cmd)
42            }
43            cmd @ DrmIoctlDropMaster => {
44                self.check_ioctl_access(DrmIoctlAccess::empty())?;
45                self.drm_drop_master(cmd)
46            }
47            // GEM ioctl cmds.
48            cmd @ DrmIoctlGemClose => {
49                self.check_gem_ioctl_access(DrmIoctlAccess::RENDER_ALLOW)?;
50                self.drm_gem_close(cmd)
51            }
52            cmd @ DrmIoctlModeCreateDumb => {
53                self.check_gem_ioctl_access(DrmIoctlAccess::empty())?;
54                self.drm_mode_create_dumb(cmd)
55            }
56            cmd @ DrmIoctlModeMapDumb => {
57                self.check_gem_ioctl_access(DrmIoctlAccess::empty())?;
58                self.drm_mode_map_dumb(cmd)
59            }
60            cmd @ DrmIoctlModeDestroyDumb => {
61                self.check_gem_ioctl_access(DrmIoctlAccess::empty())?;
62                self.drm_mode_destroy_dumb(cmd)
63            }
64            _ => {
65                ostd::warn!(
66                    "unknown ioctl minor={:?} cmd={:#x}",
67                    self.minor_type(),
68                    raw_ioctl.cmd()
69                );
70                return_errno_with_message!(Errno::ENOTTY, "the DRM ioctl command is unknown")
71            }
72        })
73    }
74
75    fn check_gem_ioctl_access(&self, required_access: DrmIoctlAccess) -> Result<()> {
76        if self.device().as_gem_ops().is_none() {
77            return_errno_with_message!(Errno::EOPNOTSUPP, "the DRM device does not support GEM");
78        }
79
80        self.check_ioctl_access(required_access)
81    }
82
83    fn check_ioctl_access(&self, required_access: DrmIoctlAccess) -> Result<()> {
84        match self.minor_type() {
85            DrmMinorType::Primary => {
86                if required_access.contains(DrmIoctlAccess::AUTH) && !self.is_authenticated() {
87                    return_errno_with_message!(
88                        Errno::EACCES,
89                        "the DRM ioctl requires an authenticated primary client"
90                    );
91                }
92            }
93            DrmMinorType::Render => {
94                if !required_access.contains(DrmIoctlAccess::RENDER_ALLOW) {
95                    return_errno_with_message!(
96                        Errno::EACCES,
97                        "the DRM ioctl is not allowed on a render node"
98                    );
99                }
100            }
101        }
102
103        if required_access.contains(DrmIoctlAccess::ROOT_ONLY) && !has_current_sys_admin() {
104            return_errno_with_message!(Errno::EACCES, "the DRM ioctl requires CAP_SYS_ADMIN");
105        }
106        if required_access.contains(DrmIoctlAccess::MASTER) && !self.is_master() {
107            return_errno_with_message!(Errno::EACCES, "the DRM client is not the current master");
108        }
109
110        Ok(())
111    }
112}
113
114bitflags::bitflags! {
115    struct DrmIoctlAccess: u32 {
116        /// Requires the client to be authenticated or the current master.
117        const AUTH = 1 << 0;
118        /// Requires the current file to be the DRM master.
119        const MASTER = 1 << 1;
120        /// Requires the caller to have `CAP_SYS_ADMIN`.
121        const ROOT_ONLY = 1 << 2;
122        /// Allows the ioctl on a render node.
123        const RENDER_ALLOW = 1 << 3;
124    }
125}
